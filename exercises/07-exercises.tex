\exercisequestion{QM-C07-S03-E03}{宇称与低阶本征函数的正交性（Exercise 7.3.3）}
\textbf{对应知识点：}\hyperref[topic:QM-C07-S03-T03]{Hermite 本征函数与归一化}。

设 $f,g\in L^2(\mathbb R)$，其中 $f$ 为偶函数、$g$ 为奇函数，两者可以取复值。
\begin{enumerate}
  \item 证明 $\int_{-\infty}^{\infty}f^*(x)g(x)\dd x=0$。
  \item 利用宇称说明谐振子的 $\psi_2$ 与 $\psi_1$ 正交。
  \item 直接积分验证 $\psi_0$ 与 $\psi_2$ 正交，说明为什么这一问不能只凭宇称作答。
\end{enumerate}
这里 $b=\sqrt{\hbar/(m\omega)}$，并使用
\begin{align}
  \psi_0(x)&=A_0\ee^{-x^2/(2b^2)},\\
  \psi_1(x)&=A_1\frac{2x}{b}\ee^{-x^2/(2b^2)},\\
  \psi_2(x)&=A_2\left(\frac{4x^2}{b^2}-2\right)\ee^{-x^2/(2b^2)},
  \qquad A_n=(b\sqrt\pi\,2^n n!)^{-1/2}.
\end{align}
可使用 $\int\ee^{-y^2}\dd y=\sqrt\pi$ 和
$\int y^2\ee^{-y^2}\dd y=\sqrt\pi/2$，积分范围均为全实轴。

\begin{checkedsolution}
\textbf{1. 先保证积分有意义。}由 Cauchy--Schwarz 不等式，
\begin{equation}
  \int_{-\infty}^{\infty}|f^*g|\dd x
  \le\left(\int|f|^2\dd x\right)^{1/2}
      \left(\int|g|^2\dd x\right)^{1/2}<\infty.
\end{equation}
因此内积绝对收敛。由于复共轭不改变偶性，
\begin{equation}
  f^*(-x)g(-x)=-f^*(x)g(x).
\end{equation}
记内积为 $I$，作代换 $x=-t$ 并处理积分上下限，有
\begin{equation}
  I=\int_{-\infty}^{\infty}f^*(-t)g(-t)\dd t
    =-\int_{-\infty}^{\infty}f^*(t)g(t)\dd t=-I.
\end{equation}
故 $\boxed{I=0}$。

\textbf{2. 奇偶态正交。}Gaussian 为偶函数，$4x^2/b^2-2$ 为偶函数而 $2x/b$ 为奇函数，所以 $\psi_2$ 偶、$\psi_1$ 奇，由第一问立刻得到
\begin{equation}
  \boxed{\braket{\psi_2}{\psi_1}=0}.
\end{equation}

\textbf{3. 两个偶态的积分抵消。}令 $y=x/b$，
\begin{align}
  \braket{\psi_0}{\psi_2}
  &=A_0^*A_2\int_{-\infty}^{\infty}
       \left(\frac{4x^2}{b^2}-2\right)\ee^{-x^2/b^2}\dd x\\
  &=bA_0^*A_2\int_{-\infty}^{\infty}(4y^2-2)\ee^{-y^2}\dd y\\
  &=bA_0^*A_2\left(4\frac{\sqrt\pi}{2}-2\sqrt\pi\right)=0.
\end{align}
所以 $\boxed{\braket{\psi_0}{\psi_2}=0}$。两个偶函数的乘积仍为偶函数，其积分\emph{未必}为零；本题为零来自正负部分的精确抵消。这也与自伴 Hamiltonian 的不同能量本征态正交相符。
\end{checkedsolution}

\exercisequestion{QM-C07-S03-E06}{无限墙约束的半谐振子（Exercise 7.3.6）}
\textbf{对应知识点：}\hyperref[topic:QM-C07-S03-T05]{宇称与边界条件}。

设 $m,\omega>0$，势能为
\begin{equation}
  V(x)=\begin{cases}
    \frac12m\omega^2x^2,&x>0,\\
    +\infty,&x\le0.
  \end{cases}
\end{equation}
利用全实轴普通谐振子的结果，求允许能量及归一化本征函数；明确边界条件、保留的宇称分支和归一化变化。

\begin{checkedsolution}
\textbf{1. 边界条件。}原点的无限墙要求
\begin{equation}
  \boxed{\phi(0)=0},\qquad
  \lim_{x\to+\infty}\phi(x)=0,\qquad
  \int_0^\infty|\phi(x)|^2\dd x<\infty.
\end{equation}
物理波函数在 $x\le0$ 为零，但墙壁\emph{不要求} $\phi'(0^+)=0$；无限墙处不能套用有限势阶跃的一阶导数连续条件。

\textbf{2. 选择满足墙壁条件的态。}普通谐振子有
\begin{equation}
  \psi_n(-x)=(-1)^n\psi_n(x),\qquad
  E_n=\left(n+\frac12\right)\hbar\omega.
\end{equation}
奇数编号态在原点为零。偶数编号态则因
$H_{2r}(0)=(-1)^r(2r)!/r!\ne0$，不满足墙壁条件。因此保留
$n=1,3,5,\ldots$。令 $n=2r+1$，得到
\begin{equation}
  \boxed{E_r=\left(2r+\frac32\right)\hbar\omega},
  \qquad r=0,1,2,\ldots.
\end{equation}
半谐振子的基态能量为 $3\hbar\omega/2$，相邻能级间隔为 $2\hbar\omega$。

\textbf{为什么这样没有漏掉其他能量？}任取半轴上的非零平方可积本征解 $\phi$，将它\emph{辅助性地奇延拓}为
\begin{equation}
  \Phi(x)=\begin{cases}\phi(x),&x>0,\\
             0,&x=0,\\-\phi(-x),&x<0.
          \end{cases}
\end{equation}
$\phi(0)=0$ 使延拓连续，两侧一阶导数也同为 $\phi'(0^+)$，故不会在二阶导数中产生额外的 delta 项。延拓满足全实轴普通谐振子的方程且平方可积，只能是其奇宇称本征态。因此上述筛选既充分又必要。这里的 $\Phi$ 只是证明用的延拓，\emph{不是}禁区内非零的物理波函数。

\textbf{3. 半轴重新归一化。}由于 $|\psi_{2r+1}|^2$ 为偶函数，
\begin{equation}
  \int_0^\infty|\psi_{2r+1}(x)|^2\dd x
    =\frac12\int_{-\infty}^{\infty}|\psi_{2r+1}(x)|^2\dd x
    =\frac12.
\end{equation}
因此限制到右半轴后应乘 $\sqrt2$，完整物理波函数为
\begin{equation}
  \boxed{\phi_r(x)=\begin{cases}
      \sqrt2\,\psi_{2r+1}(x),&x>0,\\
      0,&x\le0,
    \end{cases}}
  \qquad r=0,1,2,\ldots.
\end{equation}
特别地，半谐振子的基态来自\emph{普通谐振子第一激发态在右半轴上的部分}：
\begin{equation}
  \phi_0(x)=\frac{2x}{\pi^{1/4}b^{3/2}}\ee^{-x^2/(2b^2)}
  \quad(x>0),\qquad b=\sqrt{\frac{\hbar}{m\omega}}.
\end{equation}
它在允许区域内部没有节点。零延拓后的半谐振子物理态不是全实轴上的奇函数；“保留奇态”指借用原问题的奇态构造半轴解。
\end{checkedsolution}

\exercisequestion{QM-C07-S04-E00}{教材未编号计算示范：三次位置算符的矩阵元}
\textbf{类型与来源：}教材正文的计算示范，非独立编号 Example；印刷页 pp.~208--209。对应\hyperref[topic:QM-C07-S04-T03]{能量基中的矩阵元}。

对质量 $m>0$、角频率 $\omega>0$ 的一维谐振子，使用归一化能量本征基和相位约定
\begin{equation}
  a\ket n=\sqrt n\ket{n-1},\qquad
  a^\dagger\ket n=\sqrt{n+1}\ket{n+1},\qquad
  X=s(a+a^\dagger),\quad s=\sqrt{\frac{\hbar}{2m\omega}}.
\end{equation}
不用位置波函数积分，求 $\bra3 X^3\ket2$。

\begin{checkedsolution}
从 $\ket2$ 到 $\ket3$，三次升降操作的净效果必须为升高一层。只有两次升高、一次降低的三个有序乘积有贡献；其余五项或给出正交能级，或因降低到基态以下而为零。因此
\begin{equation}
  \bra3 X^3\ket2=s^3\bra3
  \bigl(a^\dagger a^\dagger a+a^\dagger aa^\dagger+aa^\dagger a^\dagger\bigr)\ket2.
\end{equation}
从右向左逐次作用：
\begin{align}
  a^\dagger a^\dagger a\ket2
    &=\sqrt2\,\sqrt2\,\sqrt3\ket3=2\sqrt3\ket3,\\
  a^\dagger aa^\dagger\ket2
    &=\sqrt3\,\sqrt3\,\sqrt3\ket3=3\sqrt3\ket3,\\
  aa^\dagger a^\dagger\ket2
    &=\sqrt3\,\sqrt4\,\sqrt4\ket3=4\sqrt3\ket3.
\end{align}
利用 $\braket33=1$，得到
\begin{equation}
  \boxed{\bra3 X^3\ket2
    =9\sqrt3\left(\frac{\hbar}{2m\omega}\right)^{3/2}}.
\end{equation}
三项的系数不同，不能把它们合并成 $3a^\dagger a^\dagger a$。结果量纲为长度的三次方，与 $X^3$ 一致。
\end{checkedsolution}

\exercisequestion{QM-C07-S04-E02}{教材习题：能量本征态的均值与不确定度（Exercise 7.4.2）}
\textbf{来源：}印刷页 p.~212，PDF p.~224。对归一化谐振子能量本征态 $\ket n$，求
\begin{equation}
  \expect X_n,\quad\expect P_n,\quad
  \expect{X^2}_n,\quad\expect{P^2}_n,\quad\Delta X\Delta P,
\end{equation}
其中 $\Delta X=\sqrt{\expect{X^2}_n-\expect X_n^2}$，$\Delta P$ 同理。使用前题的升降约定，并令 $P=\ii p_*(a^\dagger-a)$、$p_*=\sqrt{m\hbar\omega/2}$。

\begin{checkedsolution}
一次升降使态与原来的 $\ket n$ 正交，故
\begin{equation}
  \boxed{\expect X_n=\expect P_n=0}.
\end{equation}
两次同向升降也不能回到原能级，而先升后降与先降后升不同：
\begin{equation}
  \expect{a^2}_n=\expect{(a^\dagger)^2}_n=0,\qquad
  \expect{a^\dagger a}_n=n,\qquad\expect{aa^\dagger}_n=n+1.
\end{equation}
因此
\begin{align}
  \expect{X^2}_n
    &=s^2\expect{a^2+aa^\dagger+a^\dagger a+(a^\dagger)^2}_n\\
    &=s^2(2n+1)
     =\boxed{\frac{\hbar}{m\omega}\left(n+\frac12\right)}.
\end{align}
动量平方的展开须同时保留 $\ii^2=-1$ 和交叉项顺序：
\begin{align}
  (a^\dagger-a)^2&=(a^\dagger)^2-a^\dagger a-aa^\dagger+a^2,\\
  \expect{P^2}_n
    &=-p_*^2\expect{(a^\dagger)^2-a^\dagger a-aa^\dagger+a^2}_n\\
    &=-p_*^2[-n-(n+1)]
     =\boxed{m\hbar\omega\left(n+\frac12\right)}.
\end{align}
因为一阶矩为零，标准差为
\begin{equation}
  \Delta X=\sqrt{\frac{\hbar}{m\omega}\left(n+\frac12\right)},\qquad
  \Delta P=\sqrt{m\hbar\omega\left(n+\frac12\right)},
\end{equation}
所以
\begin{equation}
  \boxed{\Delta X\Delta P=\hbar\left(n+\frac12\right)}.
\end{equation}
在这组能量本征态中，只有 $n=0$ 达到 $\hbar/2$；这不是对\emph{所有}量子态的唯一性判断。均值为零也不代表位置、动量确定为零。

\textbf{直接推论（Exercise 7.4.3 的谐振子部分）。}由二阶矩，
\begin{equation}
  \expect T_n=\frac{\expect{P^2}_n}{2m}
    =\frac12\hbar\omega\left(n+\frac12\right),\qquad
  \expect V_n=\frac12m\omega^2\expect{X^2}_n=\expect T_n.
\end{equation}
故两者均为 $E_n/2$，与谐振子的 virial theorem（维里定理）一致。原题另有经典幂律势的部分，这里未作该部分。
\end{checkedsolution}

\exercisequestion{QM-C07-S04-E05}{教材习题：两能级叠加态与 Ehrenfest 定理（Exercise 7.4.5）}
\textbf{来源：}印刷页 p.~212，PDF p.~224。对应\hyperref[topic:QM-C07-S04-T03]{能量表象中的计算}。

取时间无关的谐振子 $H=\hbar\omega(N+I/2)$，初态为
\begin{equation}
  \ket{\psi(0)}=\frac1{\sqrt2}(\ket0+\ket1),
\end{equation}
其中 $\ket0,\ket1$ 是归一化基态和第一激发态，使用上述升降相位约定。
\begin{enumerate}
  \item 写出 $\ket{\psi(t)}$。
  \item 求 $\expect X_t,\expect P_t$，并给出初始值。
  \item 由 Ehrenfest 定理写出均值的运动方程，求解初值问题并与第二问比较；也可直接验证第二问的解满足方程和初值，再由唯一性确认一致。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 各能量分量分别积累相位。}利用 $E_0=\hbar\omega/2$、$E_1=3\hbar\omega/2$，
\begin{equation}
  \boxed{\ket{\psi(t)}
    =\frac1{\sqrt2}\left(\ee^{-\ii\omega t/2}\ket0
          +\ee^{-3\ii\omega t/2}\ket1\right)
    =\frac{\ee^{-\ii\omega t/2}}{\sqrt2}
          \left(\ket0+\ee^{-\ii\omega t}\ket1\right)}.
\end{equation}
整体相位不影响期望值；括号内的相对相位则随时间变化。

\textbf{2. 不能丢弃交叉项。}先算升降算符的期望值：
\begin{align}
  \expect a_t
    &=\frac12(\bra0+\ee^{\ii\omega t}\bra1)
          a(\ket0+\ee^{-\ii\omega t}\ket1)\\
    &=\frac12\left[\bra0a\ket0
       +\ee^{-\ii\omega t}\bra0a\ket1
       +\ee^{\ii\omega t}\bra1a\ket0+\bra1a\ket1\right]\\
    &=\frac12\ee^{-\ii\omega t},\qquad
  \expect{a^\dagger}_t=\expect a_t^*=\frac12\ee^{\ii\omega t}.
\end{align}
唯一非零项是 $\bra0a\ket1=1$ 所产生的交叉项。代入 $X=s(a+a^\dagger)$、$P=\ii p_*(a^\dagger-a)$，
\begin{align}
  \boxed{\expect X_t=s\cos\omega t},\qquad
  \boxed{\expect P_t=-p_*\sin\omega t},\\
  \boxed{\expect X_0=s=\sqrt{\frac{\hbar}{2m\omega}}},\qquad
  \boxed{\expect P_0=0}.
\end{align}
动量负号来自 $\ii(\ee^{\ii\omega t}-\ee^{-\ii\omega t})/2=-\sin\omega t$。

\textbf{3. 与经典形式的均值方程一致。}Ehrenfest 定理及对易关系给出
\begin{align}
  \frac{\dd\expect A_t}{\dd t}
    &=\frac1{\ii\hbar}\expect{[A,H]}_t
       \quad(A\text{ 不显含时间}),\\
  [X,H]&=\frac{\ii\hbar}{m}P,\qquad
  [P,H]=-\ii\hbar m\omega^2X.
\end{align}
因此，令 $x_c=\expect X_t$、$p_c=\expect P_t$，有
\begin{equation}
  \boxed{\dot x_c=p_c/m},\qquad
  \boxed{\dot p_c=-m\omega^2x_c},\qquad
  \ddot x_c+\omega^2x_c=0.
\end{equation}
初值 $x_c(0)=s$、$\dot x_c(0)=0$ 给出 $x_c=s\cos\omega t$；再由 $p_c=m\dot x_c$ 及 $m\omega s=p_*$，得到 $p_c=-p_*\sin\omega t$。也可直接求导核验第二问的两式，均满足方程和初值；线性初值问题的唯一性保证两种方法一致。

能量测量概率始终为 $1/2,1/2$，但状态不是定态，位置和动量均值因相对相位而振荡。这是谐振子均值精确满足经典方程的例子，不意味着粒子同时具有确定的位置、动量或一条确定轨迹。
\end{checkedsolution}

\noindent\begin{minipage}{\linewidth}
\exercisequestion{QM-C07-S05-E01}{教材习题：动量表象的谐振子基态（Exercise 7.5.1）}
\textbf{来源：}印刷页 p.~218，PDF p.~230。对应\hyperref[topic:QM-C07-S05-T01]{基态条件的表象转换}。

将教材式 (7.5.1) 即 $a\ket0=0$ 投影到动量基，求归一化的基态波函数 $\widetilde\psi_0(p)=\braket p0$。取 $m,\omega>0$，使用
\par
\end{minipage}
\begin{equation}
  a=\sqrt{\frac{m\omega}{2\hbar}}X
     +\frac{\ii}{\sqrt{2m\hbar\omega}}P,\qquad
  P\mapsto p,\quad X\mapsto\ii\hbar\frac{\dd}{\dd p},
\end{equation}
以及 $\braket p{p'}=\delta(p-p')$、$\int|\widetilde\psi_0(p)|^2\dd p=1$。整体相位可自由选择。

\begin{checkedsolution}
左乘 $\bra p$，将两个算符分别作用于动量波函数：
\begin{align}
  0&=\sqrt{\frac{m\omega}{2\hbar}}\,\ii\hbar
        \frac{\dd\widetilde\psi_0}{\dd p}
     +\frac{\ii p}{\sqrt{2m\hbar\omega}}\widetilde\psi_0\\
   &=\frac{\ii}{\sqrt{2m\hbar\omega}}
      \left(m\hbar\omega\frac{\dd}{\dd p}+p\right)\widetilde\psi_0.
\end{align}
两项都带 $\ii$，约去非零公因子得到
\begin{equation}
  \boxed{\frac{\dd\widetilde\psi_0}{\dd p}
       =-\frac{p}{m\hbar\omega}\widetilde\psi_0}.
\end{equation}
等价地，$\dd[\ee^{p^2/(2m\hbar\omega)}\widetilde\psi_0]/\dd p=0$，因此
\begin{equation}
  \widetilde\psi_0(p)=C\exp\left(-\frac{p^2}{2m\hbar\omega}\right).
\end{equation}
归一化给出
\begin{equation}
  1=|C|^2\int_{-\infty}^{\infty}\ee^{-p^2/(m\hbar\omega)}\dd p
   =|C|^2\sqrt{\pi m\hbar\omega}.
\end{equation}
取 $C>0$，最终
\begin{equation}
  \boxed{\widetilde\psi_0(p)=\frac1{(\pi m\hbar\omega)^{1/4}}
           \exp\left(-\frac{p^2}{2m\hbar\omega}\right)}.
\end{equation}
它的量纲为动量的 $-1/2$ 次方，与按 $\dd p$ 归一化一致。位置和动量波函数都是高斯，但宽度不能混用：由对应概率密度得
\begin{equation}
  \expect X=\expect P=0,\qquad
  \Delta X=\sqrt{\frac{\hbar}{2m\omega}},\qquad
  \Delta P=\sqrt{\frac{m\hbar\omega}{2}},\qquad
  \Delta X\Delta P=\frac\hbar2.
\end{equation}
这里宽度指概率密度的标准差，不是直接读取振幅高斯的指数分母。
\end{checkedsolution}

\noindent\begin{minipage}{\linewidth}
\exercisequestion{QM-C07-S05-E02}{教材习题：降算符与 Hermite 求导关系（Exercise 7.5.2）}
\textbf{来源：}印刷页 p.~218，PDF p.~230。对应\hyperref[topic:QM-C07-S05-T02]{升降操作与 Hermite 多项式}。

将 $a\ket n=\sqrt n\ket{n-1}$（$n\ge1$）投影到位置基，并利用教材式 (7.3.22)，证明
\begin{equation}
  H_n'(y)=2nH_{n-1}(y).
\end{equation}
\end{minipage}
其中 $y=x/b$、$b=\sqrt{\hbar/(m\omega)}$，约定
\begin{equation}
  \psi_n(x)=N_nH_n(y)\ee^{-y^2/2},\qquad
  N_n=\frac1{\pi^{1/4}\sqrt{2^n n!\,b}},\qquad
  a\mapsto\frac1{\sqrt2}\left(y+\frac{\dd}{\dd y}\right).
\end{equation}

\begin{checkedsolution}
左乘 $\bra x$，并把 $\psi_n(x)$ 写成 $y=x/b$ 的复合表达式：
\begin{equation}
  \frac1{\sqrt2}\left(y+\frac{\dd}{\dd y}\right)
       \left[N_nH_n(y)\ee^{-y^2/2}\right]
  =\sqrt n\,N_{n-1}H_{n-1}(y)\ee^{-y^2/2}.
\end{equation}
左边由乘积求导展开为
\begin{align}
  \frac{N_n}{\sqrt2}
    \left[yH_n(y)+H_n'(y)-yH_n(y)\right]\ee^{-y^2/2}
  =\frac{N_n}{\sqrt2}H_n'(y)\ee^{-y^2/2}.
\end{align}
来自乘法项和高斯导数的两个 $yH_n$ 项恰好抵消。对有限实数 $y$，高斯因子非零，故可约去，得
\begin{equation}
  H_n'(y)=\sqrt{2n}\,\frac{N_{n-1}}{N_n}H_{n-1}(y).
\end{equation}
不能省略不同能级的归一化系数之比：
\begin{equation}
  \frac{N_{n-1}}{N_n}
  =\sqrt{\frac{2^n n!\,b}{2^{n-1}(n-1)!\,b}}=\sqrt{2n}.
\end{equation}
于是
\begin{equation}
  \boxed{H_n'(y)=2nH_{n-1}(y),\qquad n\ge1}.
\end{equation}
例如 $H_1'=2=2H_0$，$H_2'=8y=4H_1$，与低阶多项式一致。抽象态的降阶关系，在位置表象中转化为 Hermite 多项式的求导关系。
\end{checkedsolution}
