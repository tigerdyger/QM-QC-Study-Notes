\exercisequestion{QM-C13-S01-E01}{教材习题 13.1.3：求归一化的氢原子态 210}
\textbf{对应知识点：}\relatedtopic{QM-C13-S01-T02}{径向递推与截断}、\relatedtopic{QM-C13-S01-T03}{氢原子波函数}。

\textbf{题目。}在固定质子、Gaussian 单位约定下，令 $a_0=\hbar^2/(m_e e^2)$、$\rho=r/(na_0)$。径向函数写成
\begin{equation}
  R_{nl}(r)=\rho^l\ee^{-\rho}\sum_{k=0}^\infty C_k\rho^k,
  \qquad
  C_{k+1}=\frac{2(k+l+1-n)}{(k+1)(k+2l+2)}C_k.
\end{equation}
从此递推关系出发，求归一化的 $\psi_{210}=R_{21}Y_1^0$。已知
\begin{equation}
  Y_1^0=\sqrt{\frac3{4\pi}}\cos\theta,\qquad
  \int|Y_1^0|^2\dd\Omega=1,
  \qquad \int_0^\infty r^q\ee^{-\beta r}\dd r
    =\frac{q!}{\beta^{q+1}}\quad(q\in\mathbb N_0,\ \beta>0).
\end{equation}
需判断截断位置、确定径向常数，并写出完整三维波函数。

\begin{checkedsolution}
\textbf{1. 截断。}代入 $n=2,l=1$，
\begin{equation}
  C_{k+1}=\frac{2k}{(k+1)(k+4)}C_k.
\end{equation}
$k=0$ 时 $C_1=0$，递推使之后各项均为零，故
\begin{equation}
  R_{21}(r)=C_0\rho\ee^{-\rho}
    =C_0\frac{r}{2a_0}\ee^{-r/(2a_0)}.
\end{equation}
这里截断的是多项式 $F$，不是将整个径向函数变成常数；前面的 $\rho$ 仍须保留。

\textbf{2. 径向归一化。}由于角向已归一化，
\begin{align}
  1&=\int_0^\infty|R_{21}|^2r^2\dd r
    =\frac{|C_0|^2}{4a_0^2}\int_0^\infty r^4\ee^{-r/a_0}\dd r\notag\\
   &=\frac{|C_0|^2}{4a_0^2}\,24a_0^5
    =6|C_0|^2a_0^3.
\end{align}
取整体相位为零、$C_0>0$，则
\begin{equation}
  C_0=\frac1{\sqrt6\,a_0^{3/2}},\qquad
  \boxed{R_{21}(r)=\frac1{2\sqrt6\,a_0^{3/2}}
     \frac r{a_0}\ee^{-r/(2a_0)}.}
\end{equation}

\textbf{3. 合成完整波函数。}乘上 $Y_1^0$，得到
\begin{equation}
  \boxed{\psi_{210}(r,\theta,\varphi)
    =\frac1{4\sqrt{2\pi}\,a_0^{3/2}}
      \frac r{a_0}\ee^{-r/(2a_0)}\cos\theta.}
\end{equation}
其量纲是长度的 $-3/2$ 次方，径向与角向的归一化积分分别为 $1$。它不含 $\varphi$，对应 $m=0$；含 $\cos\theta$ 而非球对称常数，对应 $l=1$，不能把 $m=0$ 误读成零角动量。
\end{checkedsolution}

\exercisequestion{QM-C13-S01-E02}{教材习题 13.1.5：库仑束缚态的维里定理}
\textbf{对应知识点：}\relatedtopic{QM-C13-S01-T01}{内部 Hamiltonian}、\relatedtopic{QM-C13-S01-T04}{能量与径向期望值}。

\textbf{题目。}在氢原子束缚能量本征态 $\ket{n,l,m}$ 中，取
\begin{equation}
  H=T+V,\qquad T=\frac{\mathbf P^2}{2m_e},\qquad
  V=-\frac{e^2}{r},\qquad
  \Omega=\mathbf R\cdot\mathbf P=\sum_{i=1}^3R_iP_i.
\end{equation}
$\mathbf R$ 是相对位置算符，不是质心位置，$r=|\mathbf R|$。利用定态性质及 Ehrenfest 方程
\begin{equation}
  \frac{\dd\expect\Omega}{\dd t}
    =\frac1{\ii\hbar}\expect{[\Omega,H]}=0,
\end{equation}
证明 $\expect T=-\expect V/2$。可使用 $[R_i,P_j]=\ii\hbar\delta_{ij}$、$[P_i,V]=-\ii\hbar\partial_iV$；不必代入具体波函数作积分。

\begin{checkedsolution}
\textbf{1. 动能对易子。}动量分量彼此对易，且
\begin{equation}
  [R_i,P_j^2]=[R_i,P_j]P_j+P_j[R_i,P_j]
    =2\ii\hbar\delta_{ij}P_j.
\end{equation}
用乘积法则展开，
\begin{align}
  [\Omega,T]
   &=\frac1{2m_e}\sum_{i,j}
      \left(R_i[P_i,P_j^2]+[R_i,P_j^2]P_i\right)\notag\\
   &=\frac{\ii\hbar}{m_e}\sum_iP_i^2
    =\boxed{2\ii\hbar T}.
\end{align}

\textbf{2. 势能对易子。}位置算符与 $V(\mathbf R)$ 对易，所以
\begin{equation}
  [\Omega,V]=\sum_iR_i[P_i,V]
    =-\ii\hbar\mathbf R\cdot\nabla V.
\end{equation}
对库仑势，$\nabla V=e^2\mathbf R/r^3$，所以
$\mathbf R\cdot\nabla V=e^2/r=-V$，从而
\begin{equation}
  \boxed{[\Omega,V]=\ii\hbar V.}
\end{equation}

\textbf{3. 定态中取期望。}两项相加得到
\begin{equation}
  [\Omega,H]=\ii\hbar(2T+V),\qquad
  0=\frac1{\ii\hbar}\expect{[\Omega,H]}
    =2\expect T+\expect V.
\end{equation}
因此
\begin{equation}
  \boxed{\expect T=-\frac12\expect V.}
\end{equation}
这不是算符恒等式 $2T+V=0$。任意非定态一般有非零的 $\dd\expect\Omega/\dd t$，不能直接令其为零。

\textbf{关于 $\Omega$ 的记号。}本题不要求 $\Omega$ 是可观测量；在所需矩和边界条件成立时，期望演化关系仍适用。也可采用对称化的
$D=(\mathbf R\cdot\mathbf P+\mathbf P\cdot\mathbf R)/2
=\Omega-3\ii\hbar I/2$；它与 $\Omega$ 只差常数倍单位算符，故 $[D,H]=[\Omega,H]$，结论相同。
\end{checkedsolution}

\exercisequestion{QM-C13-S02-E01}{教材习题 13.2.1：守恒矢量如何固定远近点}
\textbf{对应知识点：}\relatedtopic{QM-C13-S02-T02}{Runge--Lenz 矢量与额外守恒量}。

\textbf{题目。}在经典吸引库仑势 $V(r)=-e^2/r$ 中，考虑角动量非零的束缚轨道。已知
\begin{equation}
  E=\frac{p^2}{2m_e}-\frac{e^2}{r}<0,\qquad
  \mathbf l=\mathbf r\times\mathbf p,\qquad
  \mathbf n=\frac{\mathbf p\times\mathbf l}{m_e}-e^2\frac{\mathbf r}{r}
\end{equation}
其中能量 $E$、角动量 $\mathbf l$ 和 Runge--Lenz 矢量 $\mathbf n$ 均守恒。先取非圆轨道，记最远点、最近点的位置矢量为 $\mathbf r_{\max},\mathbf r_{\min}$，其长度为 $r_{\max}>r_{\min}>0$。
\begin{enumerate}
  \item 消去 $\mathbf l$，把 $\mathbf n$ 写成仅含 $\mathbf r,\mathbf p$ 的表达式。
  \item 说明径向转折点处为何有 $\mathbf r\cdot\mathbf p=0$，利用能量守恒证明
    \begin{equation}
      \mathbf n=\mathbf r_{\max}\left(2E+\frac{e^2}{r_{\max}}\right),\qquad
      \mathbf n=\mathbf r_{\min}\left(2E+\frac{e^2}{r_{\min}}\right).
    \end{equation}
  \item 判断 $\mathbf n$ 与两位置矢量的同向或反向关系，说明守恒性如何固定远、近点并联系闭合轨道。
  \item 单独考虑圆轨道，验证 $\mathbf n=0$。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 消去角动量。}用矢量三重积
$\mathbf a\times(\mathbf b\times\mathbf c)
=\mathbf b(\mathbf a\cdot\mathbf c)-\mathbf c(\mathbf a\cdot\mathbf b)$，有
\begin{equation}
  \mathbf p\times(\mathbf r\times\mathbf p)
   =p^2\mathbf r-(\mathbf r\cdot\mathbf p)\mathbf p,
\end{equation}
故
\begin{equation}
  \boxed{\mathbf n=\frac{p^2\mathbf r-(\mathbf r\cdot\mathbf p)\mathbf p}{m_e}
       -e^2\frac{\mathbf r}{r}.}
\end{equation}

\textbf{2. 径向极值与能量守恒。}由 $r^2=\mathbf r\cdot\mathbf r$、$\mathbf p=m_e\dot{\mathbf r}$，
\begin{equation}
  2r\dot r=2\mathbf r\cdot\dot{\mathbf r}
    =\frac2{m_e}\mathbf r\cdot\mathbf p
  \quad\Longrightarrow\quad
  \boxed{\mathbf r\cdot\mathbf p=m_er\dot r.}
\end{equation}
在远、近点 $\dot r=0$，故 $\mathbf r\cdot\mathbf p=0$。这不是 $\mathbf p=0$：角动量非零时粒子仍有切向速度。于是
\begin{equation}
  \mathbf n=\left(\frac{p^2}{m_e}-\frac{e^2}{r}\right)\mathbf r,
  \qquad \frac{p^2}{m_e}=2E+\frac{2e^2}{r}.
\end{equation}
两式结合即得
\begin{equation}
  \boxed{\mathbf n=\left(2E+\frac{e^2}{r}\right)\mathbf r
    \quad\text{（径向转折点处）}.}
\end{equation}
分别取 $r_{\max}$ 与 $r_{\min}$，便得到题目中的两式。此关系不应无条件用于轨道上的任意点。

\textbf{3. 用韦达定理确定方向。}转折点处 $\mathbf r\perp\mathbf p$，因此 $l^2=r^2p^2$，这里 $l=|\mathbf l|$ 是经典角动量大小。能量方程给出
\begin{equation}
  E=\frac{l^2}{2m_er^2}-\frac{e^2}{r}
  \quad\Longrightarrow\quad
  Er^2+e^2r-\frac{l^2}{2m_e}=0.
\end{equation}
两个根是 $r_{\min},r_{\max}$，故
\begin{equation}
  r_{\min}+r_{\max}=-\frac{e^2}{E}.
\end{equation}
消去 $E$，即可判断前面标量系数的符号：
\begin{align}
  2E+\frac{e^2}{r_{\min}}
    &=\frac{e^2(r_{\max}-r_{\min})}
      {r_{\min}(r_{\min}+r_{\max})}>0,\\
  2E+\frac{e^2}{r_{\max}}
    &=-\frac{e^2(r_{\max}-r_{\min})}
      {r_{\max}(r_{\min}+r_{\max})}<0.
\end{align}
因此 $\mathbf n$ 与 $\mathbf r_{\min}$ 同向、与 $\mathbf r_{\max}$ 反向。$\mathbf n$ 恒定便固定了这条远近点连线的朝向；它沿椭圆长轴指向近心点，不发生近心点进动。

\textbf{闭合性的逻辑。}守恒的 $E,l$ 固定两个转折半径，非零的 $\mathbf n$ 又固定其方向，所以再次到达同类转折点时，位置相同。该处 $\mathbf r\cdot\mathbf p=0$，由 $\mathbf l=\mathbf r\times\mathbf p$ 还可得 $\mathbf p=\mathbf l\times\mathbf r/r^2$，动量也相同。运动状态复现，轨道因而闭合；不只是“轨道在同一个平面内”。

\textbf{4. 圆轨道。}此时 $r$ 恒定，仍有 $\mathbf r\cdot\mathbf p=0$。库仑力提供向心力：
\begin{equation}
  \frac{p^2}{m_er}=\frac{e^2}{r^2}
  \quad\Longrightarrow\quad
  \mathbf n=\left(\frac{p^2}{m_e}-\frac{e^2}{r}\right)\mathbf r
    =\boxed{\mathbf 0}.
\end{equation}
圆没有特别的近心点方向，因此零矢量没有方向并不造成矛盾；不能在圆轨道中再使用“与 $\mathbf n$ 同向或反向”来区分远近点。
\end{checkedsolution}
