\exercisequestion{QM-C02-S01-E01}{教材练习 2.1.1：一维简谐振子}

\textbf{对应知识点：}\relatedtopic{QM-C02-S01-T02}{Euler--Lagrange 方程的完整变分推导}。

\textbf{题意。}质量为 $m$ 的物块位于无摩擦水平面上，并与劲度系数为 $k$ 的弹簧相连。写出 Lagrangian 并求运动方程。教材位置：印刷页 p.~83，PDF p.~96。

\begin{checkedsolution}
取物块偏离平衡位置的位移为 $x$。动能、势能和 Lagrangian 分别为
\[
  T=\frac12m\dot x^2,
  \qquad
  V=\frac12kx^2,
  \qquad
  \mathcal L=\frac12m\dot x^2-\frac12kx^2.
\]
所需偏导为
\[
  \frac{\partial\mathcal L}{\partial\dot x}=m\dot x,
  \qquad
  \frac{\partial\mathcal L}{\partial x}=-kx.
\]
代入 Euler--Lagrange 方程得到
\begin{equation}
  \boxed{m\ddot x+kx=0,}
  \qquad
  \omega=\sqrt{\frac{k}{m}}.
  \label{eq:QM-C02-S01-E01-oscillator}
\end{equation}
这与 Newton 方程 $m\ddot x=-kx$ 完全相同。
\end{checkedsolution}

\textbf{检查点：}势能进入 $\mathcal L=T-V$ 时带负号，因此 $\partial\mathcal L/\partial x=-kx$；最终方程的恢复力方向必须与位移相反。

\exercisequestion{QM-C02-S01-E02}{教材练习 2.1.2：耦合质量块的 Lagrangian}

\textbf{对应知识点：}\relatedtopic{QM-C02-S01-T03}{广义坐标、正则动量与循环坐标}，以及第 1.8 节的耦合振子简正模。

\textbf{题意。}两个相同质量 $m$ 的物块沿直线运动，三个相同弹簧 $k$ 依次连接墙、物块 1、物块 2 与另一侧墙。令 $x_1,x_2$ 为两个物块相对平衡位置的同向位移。写出 Lagrangian，用 Euler--Lagrange 方程推导耦合运动方程，并与第 1.8 节的 Newton 方程比较。教材位置：印刷页 p.~83，PDF p.~96。

\begin{checkedsolution}
两个物块的总动能为
\begin{equation*}
  T=\frac12m(\dot x_1^2+\dot x_2^2).
\end{equation*}
三个弹簧的有向形变量可取为 $x_1$、$x_2-x_1$ 与 $-x_2$；势能只依赖平方，因此
\begin{align*}
  V
  &=\frac12kx_1^2
    +\frac12k(x_2-x_1)^2
    +\frac12kx_2^2\\
  &=k(x_1^2+x_2^2-x_1x_2).
\end{align*}
所以
\begin{equation}
  \mathcal L
  =\frac12m(\dot x_1^2+\dot x_2^2)
  -\frac12k\left[x_1^2+(x_2-x_1)^2+x_2^2\right].
  \label{eq:QM-C02-S01-E02-lagrangian}
\end{equation}
对 $x_1$，
\[
  \frac{\partial\mathcal L}{\partial\dot x_1}=m\dot x_1,
  \qquad
  \frac{\partial\mathcal L}{\partial x_1}=-(2kx_1-kx_2),
\]
故
\[
  m\ddot x_1+2kx_1-kx_2=0.
\]
同理，对 $x_2$ 有
\[
  \frac{\partial\mathcal L}{\partial\dot x_2}=m\dot x_2,
  \qquad
  \frac{\partial\mathcal L}{\partial x_2}=-(2kx_2-kx_1),
\]
从而
\[
  m\ddot x_2+2kx_2-kx_1=0.
\]
矩阵形式为
\begin{equation}
  \boxed{
  m\begin{pmatrix}\ddot x_1\\\ddot x_2\end{pmatrix}
  +k\begin{pmatrix}2&-1\\-1&2\end{pmatrix}
  \begin{pmatrix}x_1\\x_2\end{pmatrix}=0.}
  \label{eq:QM-C02-S01-E02-coupled-equations}
\end{equation}
这正是第 1.8 节由逐个分析弹簧力得到的方程。作为独立检查，刚度矩阵在 $(1,1)^T$ 和 $(1,-1)^T$ 上的特征值分别为 $k$ 和 $3k$，故
\[
  \omega_I=\sqrt{\frac{k}{m}},
  \qquad
  \omega_{II}=\sqrt{\frac{3k}{m}},
\]
与原简正模结果一致。
\end{checkedsolution}

\textbf{检查点：}中间弹簧的形变量必须是相对位移 $x_2-x_1$。若把它误写成 $x_1+x_2$，同向等位移模式会错误地拉伸中间弹簧。

\exercisequestion{QM-C02-S01-E03}{教材练习 2.1.3：三维中心势的球坐标方程}

\textbf{对应知识点：}\relatedtopic{QM-C02-S01-T05}{三维中心势中的球坐标与角动量}。

\textbf{题意。}质量为 $m$ 的粒子在三维中心势 $V(r)$ 中运动。使用球坐标写出 Lagrangian，求 $r,\theta,\phi$ 的 Euler--Lagrange 方程，并判断循环坐标与守恒量。教材位置：印刷页 p.~83，PDF p.~96。

\begin{checkedsolution}
球坐标中的速度平方为
\[
  v^2=\dot r^2+r^2\dot\theta^2
  +r^2\sin^2\theta\,\dot\phi^2.
\]
因此
\begin{equation}
  \mathcal L
  =\frac12m\left(
    \dot r^2+r^2\dot\theta^2
    +r^2\sin^2\theta\,\dot\phi^2
  \right)-V(r).
  \label{eq:QM-C02-S01-E03-lagrangian}
\end{equation}

对 $r$，
\[
  \frac{\partial\mathcal L}{\partial\dot r}=m\dot r,
  \qquad
  \frac{\partial\mathcal L}{\partial r}
  =mr\dot\theta^2+mr\sin^2\theta\,\dot\phi^2-V'(r),
\]
所以
\begin{equation}
  \boxed{
  m\ddot r-mr\dot\theta^2
  -mr\sin^2\theta\,\dot\phi^2+V'(r)=0.}
  \label{eq:QM-C02-S01-E03-radial}
\end{equation}

对 $\theta$，
\[
  \frac{\partial\mathcal L}{\partial\dot\theta}=mr^2\dot\theta,
  \qquad
  \frac{\partial\mathcal L}{\partial\theta}
  =mr^2\sin\theta\cos\theta\,\dot\phi^2,
\]
所以
\begin{equation}
  \boxed{
  r^2\ddot\theta+2r\dot r\dot\theta
  -r^2\sin\theta\cos\theta\,\dot\phi^2=0.}
  \label{eq:QM-C02-S01-E03-theta}
\end{equation}

$\phi$ 不显含于完整的 $\mathcal L$，因此
\begin{equation}
  \boxed{
  \frac{d}{dt}\left(
    mr^2\sin^2\theta\,\dot\phi
  \right)=0,}
  \label{eq:QM-C02-S01-E03-phi}
\end{equation}
即
\begin{equation}
  \boxed{
  p_\phi=mr^2\sin^2\theta\,\dot\phi=L_z
  =\text{constant}.}
  \label{eq:QM-C02-S01-E03-lz}
\end{equation}
$r$ 显含于动能和势能，$\theta$ 通过 $\sin^2\theta$ 显含于动能，所以二者都不是循环坐标。中心势虽只直接让 $\phi$ 成为这一坐标系中的循环坐标，但完整旋转对称性仍保证整个角动量向量守恒；当前球坐标只是让 $L_z$ 的守恒最明显。
\end{checkedsolution}

\textbf{检查点：}循环坐标的判据是 $\partial\mathcal L/\partial q_i=0$，不是 $\partial V/\partial q_i=0$。$V(r)$ 与 $\theta$ 无关仍不足以让 $\theta$ 成为循环坐标，因为球坐标动能含 $\sin^2\theta$。

\exercisequestion{QM-C02-S02-E01}{自编检验题 1：均匀磁场中的正则动量}

\textbf{对应知识点：}\relatedtopic{QM-C02-S02-T03}{从 Lagrangian 完整恢复 Lorentz force}与\relatedtopic{QM-C02-S02-T04}{正则动量、机械动量与广义势}。

\textbf{题意。}在 Gaussian units 中，带电量 $q$、质量 $m$ 的粒子在 $xy$ 平面运动。取 $\phi=0$ 与 Landau gauge
\[
  \bm A=Bx\,\hat{\bm y},
\]
它对应均匀磁场 $\bm B=B\hat{\bm z}$。写出 Lagrangian，求正则动量与运动方程，并判断哪个动量分量守恒。

\begin{checkedsolution}
因为 $\bm v\cdot\bm A=Bx\dot y$，所以
\begin{equation}
  \mathcal L
  =\frac12m(\dot x^2+\dot y^2)+\frac{qB}{c}x\dot y.
  \label{eq:QM-C02-S02-E01-lagrangian}
\end{equation}
正则动量为
\begin{equation}
  p_x=m\dot x,
  \qquad
  p_y=m\dot y+\frac{qB}{c}x.
  \label{eq:QM-C02-S02-E01-canonical-momenta}
\end{equation}
对 $x$ 使用 Euler--Lagrange 方程：
\[
  \frac{d}{dt}(m\dot x)-\frac{qB}{c}\dot y=0,
\]
故
\[
  m\ddot x=\frac{qB}{c}\dot y.
\]
$y$ 不显含于 $\mathcal L$，因此是循环坐标：
\[
  \frac{d}{dt}\left(m\dot y+\frac{qB}{c}x\right)=0,
\]
即
\[
  m\ddot y=-\frac{qB}{c}\dot x.
\]
合并为
\begin{equation}
  m\ddot{\bm r}
  =\frac qc\dot{\bm r}\times B\hat{\bm z},
  \label{eq:QM-C02-S02-E01-lorentz}
\end{equation}
与 Lorentz force 一致。守恒的是 canonical momentum $p_y$，而 mechanical momentum 的 $y$ 分量 $m\dot y$ 一般不守恒；两者相差 $qBx/c$。
\end{checkedsolution}

\textbf{检查点：}循环坐标直接保证其共轭正则动量守恒，而不是自动保证相应的 Cartesian mechanical momentum 分量守恒。

\exercisequestion{QM-C02-S02-E02}{自编检验题 2：固定长度绳子的约束力}

\textbf{对应知识点：}\relatedtopic{QM-C02-S02-T05}{相互作用、坐标几何、约束与非保守力}。

\textbf{题意。}质量为 $m$ 的小球在平面内运动，并由固定在原点、长度为 $R$ 的理想不可伸长轻绳约束。系统没有普通势能。分别用消去约束坐标与 Lagrange multiplier 两种方法写出方程，并说明为什么 $V=0$ 不等于张力不存在。

\begin{checkedsolution}
若直接使用约束 $\rho=R$，唯一独立坐标为 $\phi$，约化 Lagrangian 是
\begin{equation}
  \mathcal L_{\mathrm{red}}=\frac12mR^2\dot\phi^2.
  \label{eq:QM-C02-S02-E02-reduced-lagrangian}
\end{equation}
因此
\[
  \frac{d}{dt}(mR^2\dot\phi)=0,
\]
角动量守恒。张力沿径向，而允许虚位移沿切向，所以理想张力不对允许虚位移做功；消去 $\rho$ 后，它不出现在 $\phi$ 的方程中。

若还要计算张力，则保留 $\rho$ 并引入乘子 $\lambda$：
\begin{equation}
  \mathcal L'
  =\frac12m(\dot\rho^2+\rho^2\dot\phi^2)
  -\lambda(\rho-R).
  \label{eq:QM-C02-S02-E02-multiplier-lagrangian}
\end{equation}
径向方程为
\[
  m\ddot\rho-m\rho\dot\phi^2+\lambda=0.
\]
代入 $\rho=R$ 与 $\ddot\rho=0$，得到
\begin{equation}
  \boxed{\lambda=mR\dot\phi^2.}
  \label{eq:QM-C02-S02-E02-tension}
\end{equation}
$\lambda$ 是绳子张力的大小。$V=0$ 只表示没有写入普通势能的相互作用，不能删除独立给定的约束及其反作用力。
\end{checkedsolution}

\textbf{检查点：}若约束坐标已被消去，约束力通常不会出现在剩余 Euler--Lagrange 方程中；要恢复约束力，需要保留约束并使用乘子，或另作受力分析。

\exercisequestion{QM-C02-S03-E01}{教材练习 2.3.1：分离二体 Lagrangian}

\textbf{对应知识点：}\relatedtopic{QM-C02-S03-T02}{动能分离与约化质量}。

\textbf{题意。}从两个相互作用粒子的原 Lagrangian 出发，改用质心坐标 $\bm R$ 与相对坐标 $\bm r$，推导分离后的 Lagrangian。教材位置：印刷页 p.~86，PDF p.~99。

\begin{checkedsolution}
原式为
\begin{equation}
  \mathcal L
  =\frac12m_1|\dot{\bm r}_1|^2
  +\frac12m_2|\dot{\bm r}_2|^2
  -V(\bm r_1-\bm r_2).
  \label{eq:QM-C02-S03-E01-original}
\end{equation}
令
\[
  M=m_1+m_2,
  \qquad
  \bm r=\bm r_1-\bm r_2,
  \qquad
  \bm R=\frac{m_1\bm r_1+m_2\bm r_2}{M}.
\]
逆变换及其时间导数为
\begin{align*}
  \bm r_1&=\bm R+\frac{m_2}{M}\bm r,
  &\dot{\bm r}_1&=\dot{\bm R}+\frac{m_2}{M}\dot{\bm r},\\
  \bm r_2&=\bm R-\frac{m_1}{M}\bm r,
  &\dot{\bm r}_2&=\dot{\bm R}-\frac{m_1}{M}\dot{\bm r}.
\end{align*}
因此动能为
\begin{align*}
  T={}&\frac12m_1
  \left|
    \dot{\bm R}+\frac{m_2}{M}\dot{\bm r}
  \right|^2
  +\frac12m_2
  \left|
    \dot{\bm R}-\frac{m_1}{M}\dot{\bm r}
  \right|^2\\
  ={}&\frac12(m_1+m_2)|\dot{\bm R}|^2
  +\left(
    \frac{m_1m_2}{M}-\frac{m_1m_2}{M}
  \right)\dot{\bm R}\cdot\dot{\bm r}\\
  &+\frac12\left(
    \frac{m_1m_2^2}{M^2}
    +\frac{m_2m_1^2}{M^2}
  \right)|\dot{\bm r}|^2.
\end{align*}
中间的交叉项严格为零，而最后一个系数化简为
\begin{align*}
  \frac{m_1m_2^2+m_2m_1^2}{M^2}
  &=\frac{m_1m_2(m_1+m_2)}{M^2}
   =\frac{m_1m_2}{m_1+m_2}
   \equiv\mu.
\end{align*}
同时 $V(\bm r_1-\bm r_2)=V(\bm r)$，故
\begin{equation}
  \boxed{
  \mathcal L
  =\frac12M|\dot{\bm R}|^2
  +\frac12\mu|\dot{\bm r}|^2
  -V(\bm r),
  \qquad
  \mu=\frac{m_1m_2}{m_1+m_2}.}
  \label{eq:QM-C02-S03-E01-separated}
\end{equation}
\end{checkedsolution}

\textbf{检查点：}交叉项的抵消不是偶然删项，而是来自质心坐标中的质量权重。最终式中质心项的质量必须是 $m_1+m_2$，不能误写成 $m_1+m_1$；相对项的系数则是约化质量而非总质量。

\exercisequestion{QM-C02-S04-E01}{自编检验题 1：从平稳作用量到经典路径}

\textbf{对应知识点：}
\begin{itemize}[leftmargin=*]
  \item \relatedtopic{QM-C02-S04-T01}{全局变分不等于粒子预知未来}；
  \item \relatedtopic{QM-C02-S04-T02}{Stationary 不等于 minimum}；
  \item \relatedtopic{QM-C02-S04-T03}{路径积分预告与经典路径的显现}。
\end{itemize}

\textbf{题意。}回答下列概念问题：
\begin{enumerate}[label=(\arabic*),leftmargin=*]
  \item $\delta S=0$ 是否表示经典路径的作用量一定小于所有邻近路径？
  \item 固定未来终点是否表示粒子必须预知未来？
  \item 路径积分中各路径的复振幅模长相同，为什么经典极限仍突出平稳作用量路径？
\end{enumerate}

\begin{checkedsolution}
\begin{enumerate}[label=(\arabic*),leftmargin=*]
  \item 不是。$\delta S=0$ 只表示一阶变分为零，作用量可能在该路径处取得局部最小、局部最大或鞍点；还需检查更高阶变分才能进一步分类。
  \item 不表示。未来终点是边值问题中由求解者施加的条件。变分条件导出的 Euler--Lagrange 方程是局部微分方程，也可以结合当前位置和当前速度作为初值向未来演化，无需粒子获得未来信息。
  \item 相同的是普通实作用量下 $e^{iS[x]/\hbar}$ 的模长，不是相位或概率。非平稳路径附近的微小路径改变会造成 $O(\eta)$ 的作用量变化；当 $S/\hbar\gg1$ 时，相位迅速变化，邻近贡献大多相消。经典路径满足 $\delta S=0$，其附近的作用量差从 $O(\eta^2)$ 开始，相位变化较慢，因而一簇邻近路径的振幅能够更稳定地相干叠加。
\end{enumerate}
\end{checkedsolution}

\textbf{检查点：}路径积分中的“所有路径”讨论的是复振幅求和。经典路径的出现依赖 stationary phase，而不是量子理论先把其他路径删除。

\exercisequestion{QM-C02-S05-E00}{教材讲解例题：简谐振子的 Hamiltonian}

\textbf{对应知识点：}
\begin{itemize}[leftmargin=*]
  \item \relatedtopic{QM-C02-S05-T03}{Hamilton 正则方程的完整推导}；
  \item \relatedtopic{QM-C02-S05-T05}{简谐振子与相空间轨迹}。
\end{itemize}

\textbf{题意。}一维简谐振子的 Lagrangian 为
\[
  \mathcal L=\frac12m\dot x^2-\frac12kx^2.
\]
求 Hamiltonian 与两条 Hamilton 正则方程，并验证它们恢复原来的简谐振子方程。教材位置：印刷页 pp.~89--90，PDF pp.~102--103。

\begin{checkedsolution}
正则动量和速度的逆关系是
\begin{equation}
  p=\frac{\partial\mathcal L}{\partial\dot x}=m\dot x,
  \qquad
  \dot x=\frac pm.
  \label{eq:QM-C02-S05-E00-momentum}
\end{equation}
因此
\begin{align}
  \mathcal H
  &=p\dot x-\mathcal L \notag\\
  &=m\dot x^2-\left(\frac12m\dot x^2-\frac12kx^2\right)
    =\frac12m\dot x^2+\frac12kx^2 \notag\\
  &=\boxed{\frac{p^2}{2m}+\frac12kx^2}.
  \label{eq:QM-C02-S05-E00-hamiltonian}
\end{align}
Hamilton 正则方程为
\begin{equation}
  \dot x=\frac{\partial\mathcal H}{\partial p}=\frac pm,
  \qquad
  \dot p=-\frac{\partial\mathcal H}{\partial x}=-kx.
  \label{eq:QM-C02-S05-E00-hamilton-equations}
\end{equation}
对 $p=m\dot x$ 求时间导数并代入第二式，得到
\begin{equation}
  \boxed{m\ddot x=-kx.}
  \label{eq:QM-C02-S05-E00-newton-equation}
\end{equation}
因此一条二阶 Euler--Lagrange 方程与两条一阶 Hamilton 方程给出同一动力学。
\end{checkedsolution}

\textbf{检查点：}计算 $\mathcal H=p\dot x-\mathcal L$ 后，必须使用 $\dot x=p/m$ 消去速度，才能得到以 $(x,p)$ 为独立变量的 Hamiltonian。

\exercisequestion{QM-C02-S05-E01}{教材练习 2.5.1：速度二次型与 Hamiltonian 的能量形式}

\textbf{对应知识点：}\relatedtopic{QM-C02-S05-T04}{Hamiltonian、能量与守恒必须分开判断}。

\textbf{题意。}设广义速度下的动能为
\begin{equation}
  T=\sum_{i,j}T_{ij}(q)\dot q_i\dot q_j,
  \label{eq:QM-C02-S05-E01-kinetic-energy}
\end{equation}
并且 $\mathcal L=T-V(q)$。证明
\[
  \sum_kp_k\dot q_k=2T,
\]
进而证明 $\mathcal H=T+V$。教材位置：印刷页 p.~89，PDF p.~102。

\begin{checkedsolution}
固定 $k$ 对 $\dot q_k$ 求偏导时，$\dot q_k$ 可以出现在二次型的第一个或第二个速度因子中。因此
\begin{align}
  p_k
  &=\frac{\partial\mathcal L}{\partial\dot q_k}
    =\frac{\partial T}{\partial\dot q_k} \notag\\
  &=\sum_jT_{kj}(q)\dot q_j
    +\sum_iT_{ik}(q)\dot q_i.
  \label{eq:QM-C02-S05-E01-momentum}
\end{align}
于是
\begin{align}
  \sum_kp_k\dot q_k
  &=\sum_{k,j}T_{kj}\dot q_k\dot q_j
    +\sum_{i,k}T_{ik}\dot q_i\dot q_k \notag\\
  &=T+T=2T.
  \label{eq:QM-C02-S05-E01-homogeneous-identity}
\end{align}
这个结果不需要先假定 $T_{ij}$ 对称；两项分别就是对哑指标重新命名后的原二次型。若 $T_{ij}=T_{ji}$，式~\eqref{eq:QM-C02-S05-E01-momentum} 可进一步写成
\[
  p_k=2\sum_jT_{kj}\dot q_j.
\]
教材的 $T_{ij}$ 已吸收通常二次型中的 $1/2$；若写成 $T=\frac12\sum_{ij}M_{ij}\dot q_i\dot q_j$，则 $p_k=\sum_jM_{kj}\dot q_j$。

最后，
\begin{equation}
  \boxed{
  \mathcal H=\sum_kp_k\dot q_k-\mathcal L
  =2T-(T-V)=T+V.}
  \label{eq:QM-C02-S05-E01-hamiltonian-energy}
\end{equation}
\end{checkedsolution}

\textbf{检查点：}$\mathcal H=T+V$ 在这里来自 $T$ 对速度的二次齐次性与 $V$ 的速度无关性，不是 Hamiltonian 定义本身无条件保证的恒等式。

\exercisequestion{QM-C02-S05-E02}{教材练习 2.5.2：简谐振子的相空间椭圆}

\textbf{对应知识点：}\relatedtopic{QM-C02-S05-T05}{简谐振子与相空间轨迹}。

\textbf{题意。}利用能量守恒证明简谐振子的相空间轨迹为
\[
  \left(\frac xa\right)^2+\left(\frac pb\right)^2=1,
\]
求 $a^2,b^2$，并判断轨迹在右端点的运动方向。教材位置：印刷页 p.~90，PDF p.~103。

\begin{checkedsolution}
固定能量 $E$ 满足
\begin{equation}
  E=\frac{p^2}{2m}+\frac12kx^2.
  \label{eq:QM-C02-S05-E02-energy}
\end{equation}
两边除以 $E$ 并整理：
\begin{equation}
  \boxed{
  \frac{x^2}{2E/k}+\frac{p^2}{2mE}=1,}
  \label{eq:QM-C02-S05-E02-ellipse}
\end{equation}
所以
\begin{equation}
  \boxed{
  a^2=\frac{2E}{k},
  \qquad
  b^2=2mE.}
  \label{eq:QM-C02-S05-E02-semiaxes}
\end{equation}
不能漏掉 $E$：$2/k$ 不具有长度平方的量纲，而 $2E/k$ 才有。

在右端点 $(a,0)$，
\[
  \dot x=\frac pm=0,
  \qquad
  \dot p=-kx=-ka<0,
\]
所以相空间点下一瞬间向下运动。整条椭圆是同一个能量 $E$ 的等值线；椭圆上的不同点对应同一振子运动过程中的不同状态。
\end{checkedsolution}

\textbf{检查点：}能量等值线只给出允许轨迹的几何形状；Hamilton 正则方程还给出沿轨迹运动的方向与速率。

\exercisequestion{QM-C02-S05-E03}{教材练习 2.5.3：耦合振子的 Hamilton 形式}

\textbf{对应知识点：}\relatedtopic{QM-C02-S05-T03}{Hamilton 正则方程的完整推导}。

\textbf{题意。}两个质量均为 $m$ 的物块由三个劲度系数均为 $k$ 的弹簧连接，两端固定。设位移为 $x_1,x_2$，则
\begin{equation}
  \mathcal L
  =\frac12m(\dot x_1^2+\dot x_2^2)
  -\frac12k\left[x_1^2+(x_2-x_1)^2+x_2^2\right].
  \label{eq:QM-C02-S05-E03-lagrangian}
\end{equation}
用 Hamilton 形式重新推导耦合运动方程。教材位置：印刷页 p.~90，PDF p.~103；本题要求重做练习 2.1.2。

\begin{checkedsolution}
正则动量及逆关系为
\begin{equation}
  p_1=m\dot x_1,
  \qquad
  p_2=m\dot x_2,
  \qquad
  \dot x_1=\frac{p_1}{m},
  \qquad
  \dot x_2=\frac{p_2}{m}.
  \label{eq:QM-C02-S05-E03-momenta}
\end{equation}
因此
\begin{equation}
  \boxed{
  \mathcal H
  =\frac{p_1^2}{2m}+\frac{p_2^2}{2m}
  +\frac12k\left[x_1^2+(x_2-x_1)^2+x_2^2\right].}
  \label{eq:QM-C02-S05-E03-hamiltonian}
\end{equation}
四条 Hamilton 正则方程是
\begin{align}
  \dot x_1&=\frac{p_1}{m},
  &\dot x_2&=\frac{p_2}{m},
  \label{eq:QM-C02-S05-E03-coordinate-equations}\\
  \dot p_1&=-2kx_1+kx_2,
  &\dot p_2&=-2kx_2+kx_1.
  \label{eq:QM-C02-S05-E03-momentum-equations}
\end{align}
利用 $\dot p_i=m\ddot x_i$，可恢复
\begin{equation}
  \boxed{
  m\ddot x_1+2kx_1-kx_2=0,
  \qquad
  m\ddot x_2+2kx_2-kx_1=0.}
  \label{eq:QM-C02-S05-E03-coupled-equations}
\end{equation}
这与第 2.1 节从 Euler--Lagrange 方程所得结果一致。
\end{checkedsolution}

\textbf{检查点：}两个二阶方程改写为四个一阶方程后，自由度没有增加；初始数据只是由 $(x_i,\dot x_i)$ 改写为 $(x_i,p_i)$。

\exercisequestion{QM-C02-S05-E04}{教材练习 2.5.4：二体问题的 Hamiltonian}

\textbf{对应知识点：}\relatedtopic{QM-C02-S05-T03}{Hamilton 正则方程的完整推导}，并联系第 2.3 节的质心与相对坐标分离。

\textbf{题意。}二体问题在质心坐标 $\bm R$ 与相对坐标 $\bm r$ 下的 Lagrangian 为
\begin{equation}
  \mathcal L
  =\frac12M|\dot{\bm R}|^2
  +\frac12\mu|\dot{\bm r}|^2
  -V(\bm r),
  \qquad
  M=m_1+m_2,
  \quad
  \mu=\frac{m_1m_2}{m_1+m_2}.
  \label{eq:QM-C02-S05-E04-lagrangian}
\end{equation}
推导相应 Hamiltonian，并说明质心动量为什么守恒。教材位置：印刷页 p.~90，PDF p.~103。

\begin{checkedsolution}
质心坐标与相对坐标的共轭动量分别为
\begin{equation}
  \bm P=M\dot{\bm R},
  \qquad
  \bm p=\mu\dot{\bm r},
  \label{eq:QM-C02-S05-E04-momenta}
\end{equation}
所以
\[
  \dot{\bm R}=\frac{\bm P}{M},
  \qquad
  \dot{\bm r}=\frac{\bm p}{\mu}.
\]
代入 Legendre 变换：
\begin{align}
  \mathcal H
  &=\bm P\cdot\dot{\bm R}
    +\bm p\cdot\dot{\bm r}-\mathcal L \notag\\
  &=\frac{|\bm P|^2}{M}+\frac{|\bm p|^2}{\mu}
    -\left(
      \frac{|\bm P|^2}{2M}
      +\frac{|\bm p|^2}{2\mu}
      -V(\bm r)
    \right) \notag\\
  &=\boxed{
    \frac{|\bm P|^2}{2M}
    +\frac{|\bm p|^2}{2\mu}
    +V(\bm r)}.
  \label{eq:QM-C02-S05-E04-hamiltonian}
\end{align}
相对动量项的分母必须是 $2\mu$，因为 $\bm p$ 与 $\bm r$ 共轭；不能误写成任一原粒子的质量。

由于 $\mathcal H$ 不显含 $\bm R$，
\begin{equation}
  \dot{\bm P}
  =-\frac{\partial\mathcal H}{\partial\bm R}=0.
  \label{eq:QM-C02-S05-E04-total-momentum}
\end{equation}
因此 $\bm P$ 守恒，且 $\dot{\bm R}=\bm P/M$ 为常量。物理结论是质心静止或做匀速直线运动；两个粒子仍可围绕质心发生非平凡的相对运动。
\end{checkedsolution}

\textbf{检查点：}质心运动和相对运动在 Hamiltonian 中分别由 $(\bm R,\bm P)$ 与 $(\bm r,\bm p)$ 描述。总动量守恒约束的是质心，不会冻结内部相对自由度。

\exercisequestion{QM-C02-S06-E01}{自编检验题 1：Landau gauge 下的均匀磁场}

\textbf{对应知识点：}
\begin{itemize}[leftmargin=*]
  \item \relatedtopic{QM-C02-S06-T01}{正则动量、机械动量与 Legendre 变换}；
  \item \relatedtopic{QM-C02-S06-T03}{由 Hamilton 正则方程恢复 Lorentz force}；
  \item \relatedtopic{QM-C02-S06-T04}{规范变换下什么保持不变}。
\end{itemize}

\textbf{题意。}在 Gaussian units 中，质量为 $m$、电荷为 $q$ 的粒子在 $xy$ 平面运动。取
\begin{equation}
  \phi=0,
  \qquad
  \bm A=Bx\,\hat{\bm y},
  \label{eq:QM-C02-S06-E01-landau-gauge}
\end{equation}
它对应均匀磁场 $\bm B=B\hat{\bm z}$。求二维 Hamiltonian，找出循环坐标与守恒的正则动量，写出四条 Hamilton 正则方程，并恢复 Lorentz-force equations。

\begin{checkedsolution}
令
\[
  \alpha\equiv\frac{qB}{c}.
\]
因为 $\bm v\cdot\bm A=Bx\dot y$，Lagrangian 为
\begin{equation}
  \mathcal L
  =\frac12m(\dot x^2+\dot y^2)+\alpha x\dot y.
  \label{eq:QM-C02-S06-E01-lagrangian}
\end{equation}
正则动量及速度的逆关系是
\begin{equation}
  p_x=m\dot x,
  \qquad
  p_y=m\dot y+\alpha x,
  \qquad
  \dot x=\frac{p_x}{m},
  \qquad
  \dot y=\frac{p_y-\alpha x}{m}.
  \label{eq:QM-C02-S06-E01-momenta}
\end{equation}
Legendre 变换给出
\begin{align}
  \mathcal H
  &=p_x\dot x+p_y\dot y-\mathcal L \notag\\
  &=\boxed{
    \frac{p_x^2}{2m}
    +\frac1{2m}(p_y-\alpha x)^2}.
  \label{eq:QM-C02-S06-E01-hamiltonian}
\end{align}
这正是
\[
  \mathcal H
  =\frac1{2m}
  \left|\bm p-\frac qc\bm A\right|^2
\]
在当前规范中的二维形式。

$\mathcal H$ 显含 $x$，但不显含 $y$，所以 $y$ 是循环坐标，并且
\begin{equation}
  \dot p_y
  =-\frac{\partial\mathcal H}{\partial y}=0.
  \label{eq:QM-C02-S06-E01-conserved-py}
\end{equation}
守恒的是
\[
  p_y=m\dot y+\frac{qB}{c}x,
\]
而不是单独的机械动量分量 $m\dot y$。

四条 Hamilton 正则方程为
\begin{align}
  \dot x
  &=\frac{\partial\mathcal H}{\partial p_x}
    =\frac{p_x}{m},
  &
  \dot y
  &=\frac{\partial\mathcal H}{\partial p_y}
    =\frac{p_y-\alpha x}{m},
  \label{eq:QM-C02-S06-E01-coordinate-equations}\\
  \dot p_x
  &=-\frac{\partial\mathcal H}{\partial x}
    =\frac{\alpha}{m}(p_y-\alpha x)
    =\alpha\dot y,
  &
  \dot p_y
  &=-\frac{\partial\mathcal H}{\partial y}=0.
  \label{eq:QM-C02-S06-E01-momentum-equations}
\end{align}
由 $p_x=m\dot x$ 得
\begin{equation}
  m\ddot x=\dot p_x
  =\frac{qB}{c}\dot y.
  \label{eq:QM-C02-S06-E01-x-equation}
\end{equation}
另一方面，对 $p_y=m\dot y+\alpha x$ 求全时间导数，并使用 $\dot p_y=0$：
\begin{equation}
  0=m\ddot y+\alpha\dot x,
  \qquad
  m\ddot y=-\frac{qB}{c}\dot x.
  \label{eq:QM-C02-S06-E01-y-equation}
\end{equation}
因此
\begin{equation}
  \boxed{
  m\ddot x=\frac{qB}{c}\dot y,
  \qquad
  m\ddot y=-\frac{qB}{c}\dot x.}
  \label{eq:QM-C02-S06-E01-lorentz-equations}
\end{equation}
因为
\[
  (\dot x,\dot y,0)\times(0,0,B)
  =(B\dot y,-B\dot x,0),
\]
这正是
\[
  m\dot{\bm v}=\frac qc\bm v\times\bm B
\]
的两个非零分量。
\end{checkedsolution}

\textbf{检查点：}$p_y$ 守恒不表示 $m\dot y$ 守恒，因为矢势贡献 $\alpha x$ 会随轨迹变化。能否从 Hamiltonian 直接识别守恒量取决于所选规范中的循环坐标，但最终的机械轨迹与规范选择无关。

\exercisequestion{QM-C02-S07-E01}{教材习题 2.7.1：Poisson 括号的代数性质}

\textbf{对应知识点：}\relatedtopic{QM-C02-S07-T02}{Poisson 括号的代数与基本括号}。

\textbf{题意。}由 Poisson bracket 的定义验证
\begin{align}
  \{F,G\}&=-\{G,F\},
  \label{eq:QM-C02-S07-E01-antisymmetry}\\
  \{F,G+K\}&=\{F,G\}+\{F,K\},
  \label{eq:QM-C02-S07-E01-additivity}\\
  \{F,GK\}&=\{F,G\}K+G\{F,K\}.
  \label{eq:QM-C02-S07-E01-product}
\end{align}

\begin{checkedsolution}
由定义
\[
  \{F,G\}
  =\sum_i\left(
    \frac{\partial F}{\partial q_i}
    \frac{\partial G}{\partial p_i}
    -
    \frac{\partial F}{\partial p_i}
    \frac{\partial G}{\partial q_i}
  \right).
\]
交换 $F,G$ 后，
\begin{align}
  \{G,F\}
  &=\sum_i\left(
    \frac{\partial G}{\partial q_i}
    \frac{\partial F}{\partial p_i}
    -
    \frac{\partial G}{\partial p_i}
    \frac{\partial F}{\partial q_i}
  \right) \notag\\
  &=-\{F,G\},
\end{align}
故反对称性成立，并立即推出 $\{F,F\}=0$。

对第二个变量的和使用偏导线性：
\begin{align}
  \{F,G+K\}
  &=\sum_i\left[
    \frac{\partial F}{\partial q_i}
    \left(
      \frac{\partial G}{\partial p_i}
      +\frac{\partial K}{\partial p_i}
    \right)
    -
    \frac{\partial F}{\partial p_i}
    \left(
      \frac{\partial G}{\partial q_i}
      +\frac{\partial K}{\partial q_i}
    \right)
  \right] \notag\\
  &=\{F,G\}+\{F,K\}.
\end{align}

最后使用普通偏导的乘积法则：
\[
  \frac{\partial(GK)}{\partial p_i}
  =\frac{\partial G}{\partial p_i}K
   +G\frac{\partial K}{\partial p_i},
  \qquad
  \frac{\partial(GK)}{\partial q_i}
  =\frac{\partial G}{\partial q_i}K
   +G\frac{\partial K}{\partial q_i}.
\]
代入括号定义并按 $K$ 与 $G$ 分组：
\begin{align}
  \{F,GK\}
  ={}&
  \sum_i\left(
    \frac{\partial F}{\partial q_i}
    \frac{\partial G}{\partial p_i}
    -
    \frac{\partial F}{\partial p_i}
    \frac{\partial G}{\partial q_i}
  \right)K \notag\\
  &+G\sum_i\left(
    \frac{\partial F}{\partial q_i}
    \frac{\partial K}{\partial p_i}
    -
    \frac{\partial F}{\partial p_i}
    \frac{\partial K}{\partial q_i}
  \right) \notag\\
  ={}&\{F,G\}K+G\{F,K\}.
\end{align}
三条性质均得证。它们分别对应对易子的反对称性、线性与 Leibniz rule，但 Poisson bracket 仍是经典相空间上的微分运算。
\end{checkedsolution}

\textbf{检查点：}乘积法则直接继承自普通偏导的乘积法则；不要把 $\{F,GK\}$ 错写成 $\{F,G\}\{F,K\}$。

\exercisequestion{QM-C02-S07-E02}{教材习题 2.7.5：极坐标变量是正则变量}

\textbf{对应知识点：}\relatedtopic{QM-C02-S07-T04}{一般正则变换及其判据}。

\textbf{题意。}在二维相空间中定义
\begin{equation}
  \rho=\sqrt{x^2+y^2},
  \qquad
  \phi=\tan^{-1}\frac yx,
  \label{eq:QM-C02-S07-E02-polar-coordinates}
\end{equation}
以及
\begin{equation}
  p_\rho=\frac{xp_x+yp_y}{\rho},
  \qquad
  p_\phi=xp_y-yp_x.
  \label{eq:QM-C02-S07-E02-polar-momenta}
\end{equation}
在 $\rho>0$ 且选定连续角坐标分支的区域，证明
\[
  (x,y,p_x,p_y)
  \longmapsto
  (\rho,\phi,p_\rho,p_\phi)
\]
是 canonical transformation。

\begin{checkedsolution}
在旧变量中，Poisson bracket 是
\begin{align}
  \{F,G\}
  ={}&
  \frac{\partial F}{\partial x}
  \frac{\partial G}{\partial p_x}
  -
  \frac{\partial F}{\partial p_x}
  \frac{\partial G}{\partial x} \notag\\
  &+
  \frac{\partial F}{\partial y}
  \frac{\partial G}{\partial p_y}
  -
  \frac{\partial F}{\partial p_y}
  \frac{\partial G}{\partial y}.
  \label{eq:QM-C02-S07-E02-old-bracket}
\end{align}
所需坐标偏导为
\begin{equation}
  \rho_x=\frac x\rho,
  \quad
  \rho_y=\frac y\rho,
  \quad
  \phi_x=-\frac y{\rho^2},
  \quad
  \phi_y=\frac x{\rho^2}.
  \label{eq:QM-C02-S07-E02-coordinate-derivatives}
\end{equation}
动量偏导为
\begin{equation}
  (p_\rho)_{p_x}=\frac x\rho,
  \quad
  (p_\rho)_{p_y}=\frac y\rho,
  \quad
  (p_\phi)_{p_x}=-y,
  \quad
  (p_\phi)_{p_y}=x.
  \label{eq:QM-C02-S07-E02-momentum-derivatives}
\end{equation}

由于 $\rho,\phi$ 都不依赖动量，立刻有
\begin{equation}
  \{\rho,\phi\}=0.
  \label{eq:QM-C02-S07-E02-coordinate-bracket}
\end{equation}
两组共轭括号分别为
\begin{align}
  \{\rho,p_\rho\}
  &=\frac x\rho\frac x\rho
    +\frac y\rho\frac y\rho
    =\frac{x^2+y^2}{\rho^2}=1,
  \label{eq:QM-C02-S07-E02-radial-pair}\\
  \{\phi,p_\phi\}
  &=\left(-\frac y{\rho^2}\right)(-y)
    +\frac x{\rho^2}x
    =\frac{x^2+y^2}{\rho^2}=1.
  \label{eq:QM-C02-S07-E02-angular-pair}
\end{align}
两项交叉括号为
\begin{align}
  \{\rho,p_\phi\}
  &=\frac x\rho(-y)+\frac y\rho x=0,
  \label{eq:QM-C02-S07-E02-rho-pphi}\\
  \{\phi,p_\rho\}
  &=\left(-\frac y{\rho^2}\right)\frac x\rho
    +\frac x{\rho^2}\frac y\rho=0.
  \label{eq:QM-C02-S07-E02-phi-prho}
\end{align}

最后验证动量之间的括号。记
\[
  N=xp_x+yp_y,
  \qquad
  p_\rho=\frac N\rho.
\]
则
\begin{align}
  \frac{\partial p_\rho}{\partial x}
  &=\frac{p_x}{\rho}-\frac{Nx}{\rho^3},
  &
  \frac{\partial p_\rho}{\partial y}
  &=\frac{p_y}{\rho}-\frac{Ny}{\rho^3},
  \label{eq:QM-C02-S07-E02-prho-position-derivatives}\\
  \frac{\partial p_\phi}{\partial x}
  &=p_y,
  &
  \frac{\partial p_\phi}{\partial y}
  &=-p_x.
  \label{eq:QM-C02-S07-E02-pphi-position-derivatives}
\end{align}
因此
\begin{align}
  \{p_\rho,p_\phi\}
  ={}&
  \left(\frac{p_x}{\rho}-\frac{Nx}{\rho^3}\right)(-y)
  -\frac x\rho p_y \notag\\
  &+
  \left(\frac{p_y}{\rho}-\frac{Ny}{\rho^3}\right)x
  -\frac y\rho(-p_x) \notag\\
  ={}&0.
  \label{eq:QM-C02-S07-E02-momentum-bracket}
\end{align}
至此六组基本括号满足
\begin{equation}
  \boxed{
  \{\rho,\phi\}=\{p_\rho,p_\phi\}=0,
  \quad
  \{\rho,p_\rho\}=\{\phi,p_\phi\}=1,
  \quad
  \{\rho,p_\phi\}=\{\phi,p_\rho\}=0.}
  \label{eq:QM-C02-S07-E02-canonical-result}
\end{equation}
所以极坐标变量构成一组 canonical coordinates。

对于
\[
  \mathcal L
  =\frac12m(\dot\rho^2+\rho^2\dot\phi^2)-V(\rho,\phi),
\]
相应动量具有直接物理意义：
\begin{equation}
  p_\rho=m\dot\rho,
  \qquad
  p_\phi=m\rho^2\dot\phi=L_z.
  \label{eq:QM-C02-S07-E02-physical-momenta}
\end{equation}
若 $V$ 不依赖 $\phi$，则 $\phi$ 是循环坐标，$p_\phi=L_z$ 守恒。
\end{checkedsolution}

\textbf{检查点：}极坐标变换在原点没有定义，角坐标还需选择局部分支；“六组括号正确”证明的是局部正则性，不能自动消除全局坐标奇点。
