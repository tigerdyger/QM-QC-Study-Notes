\exercisequestion{QM-C12-S01-E01}{教材习题 12.1.1：从波函数平移识别生成元}
\textbf{对应知识点：}\relatedtopic{QM-C12-S01-T01}{方向上的动量分量}。题意统一记号重述，缺页及转录来源说明见\hyperref[section:QM-C12-S01]{正文}。

\textbf{题目。}设 $\mathbf n=(n_x,n_y)$ 是固定单位向量，$\psi(x,y)$ 足够光滑。主动平移位移 $\epsilon\mathbf n$ 后，波函数为
\begin{equation}
  \psi_\epsilon(x,y)=\psi(x-\epsilon n_x,y-\epsilon n_y).
\end{equation}
利用 $P_x=-\ii\hbar\partial_x$、$P_y=-\ii\hbar\partial_y$，将一阶展开与
$\psi_\epsilon=(I-\ii\epsilon G_{\mathbf n}/\hbar)\psi+O(\epsilon^2)$ 比较，验证沿 $\mathbf n$ 方向的无穷小平移生成元为 $\mathbf n\cdot\mathbf P$。

\begin{checkedsolution}
对两个自变量同时作一阶 Taylor 展开：
\begin{align}
  \psi_\epsilon(x,y)
  &=\psi(x,y)-\epsilon n_x\partial_x\psi(x,y)
    -\epsilon n_y\partial_y\psi(x,y)+O(\epsilon^2)\notag\\
  &=\left[I-\epsilon(n_x\partial_x+n_y\partial_y)\right]\psi
    +O(\epsilon^2)\notag\\
  &=\left[I-\frac{\ii\epsilon}{\hbar}(n_xP_x+n_yP_y)\right]\psi
    +O(\epsilon^2).
\end{align}
其中 $(-\ii)(-\ii)=-1$ 保证符号一致。对任意上述波函数比较生成元定义，得到
\begin{equation}
  \boxed{G_{\mathbf n}=n_xP_x+n_yP_y=\mathbf n\cdot\mathbf P}.
\end{equation}
\end{checkedsolution}

\exercisequestion{QM-C12-S02-E01}{教材习题 12.2.2：核对角动量的四条对易关系}
\textbf{对应知识点：}\relatedtopic{QM-C12-S02-T02}{分量变换与对易关系}。

\textbf{题目。}教材要求由旋转所需的对易关系确定 $L_z$，或至少验证 $L_z=XP_y-YP_x$ 与之相容。本题采用后一要求。已知
\begin{equation}
  [X,P_x]=[Y,P_y]=\ii\hbar I,\qquad
  [X,Y]=[P_x,P_y]=[X,P_y]=[Y,P_x]=0,
\end{equation}
验证
\begin{equation}
  [X,L_z]=-\ii\hbar Y,\quad [Y,L_z]=\ii\hbar X,\quad
  [P_x,L_z]=-\ii\hbar P_y,\quad [P_y,L_z]=\ii\hbar P_x.
\end{equation}

\begin{checkedsolution}
使用 $[A,BC]=[A,B]C+B[A,C]$。对位置分量，
\begin{align}
  [X,L_z]&=[X,X]P_y+X[X,P_y]-[X,Y]P_x-Y[X,P_x]\notag\\
         &=0+0-0-\ii\hbar Y=\boxed{-\ii\hbar Y},\\
  [Y,L_z]&=[Y,X]P_y+X[Y,P_y]-[Y,Y]P_x-Y[Y,P_x]\notag\\
         &=0+\ii\hbar X-0-0=\boxed{\ii\hbar X}.
\end{align}
对动量分量，注意对易子反对称性给出 $[P_x,X]=[P_y,Y]=-\ii\hbar I$，于是
\begin{align}
  [P_x,L_z]&=[P_x,X]P_y+X[P_x,P_y]-[P_x,Y]P_x-Y[P_x,P_x]\notag\\
           &=-\ii\hbar P_y+0-0-0=\boxed{-\ii\hbar P_y},\\
  [P_y,L_z]&=[P_y,X]P_y+X[P_y,P_y]-[P_y,Y]P_x-Y[P_y,P_x]\notag\\
           &=0+0-(-\ii\hbar P_x)-0=\boxed{\ii\hbar P_x}.
\end{align}
四条关系均成立。将它们用于 $U^\dagger A U=A+\ii\epsilon[L_z,A]/\hbar+O(\epsilon^2)$，位置和动量都按二维无穷小旋转规律变换。

\textbf{严格性补充。}这验证了一致性，却没有单凭对易关系唯一确定生成元：$L_z+cI$（$c$ 为实常数，具有角动量量纲）也满足同样的四条关系。它使旋转算符多出 $\ee^{-\ii\theta c/\hbar}$；正文选用不附加该整体相位的纯轨道旋转约定。
\end{checkedsolution}

\exercisequestion{QM-C12-S02-E02}{教材习题 12.2.3：用两种方法得到极坐标角动量}
\textbf{对应知识点：}\relatedtopic{QM-C12-S02-T02}{有限旋转与极坐标}。

\textbf{题目。}证明 $L_z=-\ii\hbar\partial_\varphi$，其中对 $\varphi$ 的偏导保持 $\rho$ 不变。要求分别使用：
\begin{enumerate}
  \item 坐标变换法：由 $L_z=-\ii\hbar(x\partial_y-y\partial_x)$ 和 $x=\rho\cos\varphi$、$y=\rho\sin\varphi$ 推出结论；
  \item 直接法：将主动小角度旋转 $\psi_\epsilon(\rho,\varphi)=\psi(\rho,\varphi-\epsilon)$ 与生成元定义 $U(\epsilon)=I-\ii\epsilon L_z/\hbar+O(\epsilon^2)$ 比较。
\end{enumerate}

\begin{checkedsolution}
\textbf{方法一：链式法则。}取任意足够光滑的 $f(x,y)$，定义
$F(\rho,\varphi)=f(\rho\cos\varphi,\rho\sin\varphi)$。固定 $\rho$ 时，
\begin{equation}
  \frac{\partial x}{\partial\varphi}=-\rho\sin\varphi=-y,\qquad
  \frac{\partial y}{\partial\varphi}=\rho\cos\varphi=x.
\end{equation}
于是
\begin{align}
  \frac{\partial F}{\partial\varphi}
  &=\frac{\partial f}{\partial x}\frac{\partial x}{\partial\varphi}
    +\frac{\partial f}{\partial y}\frac{\partial y}{\partial\varphi}\notag\\
  &=-y\frac{\partial f}{\partial x}+x\frac{\partial f}{\partial y}
   =(x\partial_y-y\partial_x)f.
\end{align}
因此，同一微分算符在极坐标表示中为
\begin{equation}
  \boxed{L_z=-\ii\hbar(x\partial_y-y\partial_x)
    =-\ii\hbar\partial_\varphi}.
\end{equation}

\textbf{方法二：无穷小旋转。}主动逆时针旋转后，当前极角处的振幅来自原来的 $\varphi-\epsilon$，所以
\begin{align}
  \psi_\epsilon(\rho,\varphi)
  &=\psi(\rho,\varphi-\epsilon)\notag\\
  &=\psi(\rho,\varphi)-\epsilon\partial_\varphi\psi(\rho,\varphi)
    +O(\epsilon^2).
\end{align}
另一方面，生成元定义给出
\begin{equation}
  \psi_\epsilon=\psi-\frac{\ii\epsilon}{\hbar}L_z\psi+O(\epsilon^2).
\end{equation}
对任意上述波函数比较一阶系数，
\begin{equation}
  -\partial_\varphi=-\frac{\ii}{\hbar}L_z
  \quad\Longrightarrow\quad
  \boxed{L_z=-\ii\hbar\partial_\varphi}.
\end{equation}
两种方法分别说明同一算符如何改变坐标表达，以及如何直接由几何变换识别生成元。角导数对应旋转，正如 $P_x=-\ii\hbar\partial_x$ 的位置导数对应平移。
\end{checkedsolution}

\exercisequestion{QM-C12-S03-E01}{教材习题 12.3.3：从角向展开求测量概率}
\textbf{对应知识点：}\relatedtopic{QM-C12-S03-T02}{角向基、径向简并与测量概率}。

\textbf{题目。}平面粒子的归一化波函数为
\begin{equation}
  \psi(\rho,\varphi)=A\ee^{-\rho^2/(2\Delta^2)}\cos^2\varphi,
  \qquad \Delta>0,\quad 0\leq\rho<\infty,\quad 0\leq\varphi<2\pi,
\end{equation}
其中 $A$ 是归一化常数，积分测度为 $\rho\,\dd\rho\,\dd\varphi$。将角向部分展开为 $\Phi_m(\varphi)=\ee^{\ii m\varphi}/\sqrt{2\pi}$，求 $L_z$ 的可能测量结果及概率，并说明为何本题无需显式计算径向积分。

\begin{checkedsolution}
\textbf{1. 识别角向分量。}由倍角公式和 Euler 公式，
\begin{align}
  \cos^2\varphi
  &=\frac{1+\cos2\varphi}{2}
   =\frac12+\frac14\ee^{2\ii\varphi}+\frac14\ee^{-2\ii\varphi}\notag\\
  &=\sqrt{2\pi}\left(\frac12\Phi_0
                   +\frac14\Phi_2+\frac14\Phi_{-2}\right).
\end{align}
只有 $m=0,+2,-2$ 三个分量，故可能测得 $0,+2\hbar,-2\hbar$。复指数前的 $1/2,1/4,1/4$ 不是已归一化的态系数，不能直接将其平方当成最终概率。

\textbf{2. 径向因子为何能约去。}记 $R(\rho)=A\ee^{-\rho^2/(2\Delta^2)}$、$I_R=\int_0^\infty\rho|R|^2\,\dd\rho$，并令
\begin{equation}
  a_0=\frac12,\qquad a_2=a_{-2}=\frac14.
\end{equation}
相对于归一化基 $\Phi_m$，各径向系数为 $f_m=\sqrt{2\pi}\,a_mR$。由角向正交性，
\begin{equation}
  \|\psi\|^2=2\pi I_R\sum_m|a_m|^2,\qquad
  \|\text{第 }m\text{ 个投影分量}\|^2=2\pi I_R|a_m|^2.
\end{equation}
所以归一化后的概率为
\begin{equation}
  P(m)=\frac{2\pi I_R|a_m|^2}
             {2\pi I_R\sum_n|a_n|^2}
      =\frac{|a_m|^2}{\sum_n|a_n|^2}.
\end{equation}
约去的是所有角向分量\emph{共同}的径向积分与基归一化因子，不是径向自由度一般都无关。

\textbf{3. 概率结果。}系数模平方之和为
\begin{equation}
  \sum_m|a_m|^2=\frac14+\frac1{16}+\frac1{16}=\frac38.
\end{equation}
于是
\begin{equation}
  \boxed{P(L_z=0)=\frac23,\qquad
         P(L_z=2\hbar)=\frac16,\qquad
         P(L_z=-2\hbar)=\frac16.}
\end{equation}
三项相加为 $1$；等价的归一化角向态是
$\sqrt{2/3}\,\Phi_0+(\Phi_2+\Phi_{-2})/\sqrt6$。因此本题不必先求出 $A$，结果也不依赖 $\Delta$。
\end{checkedsolution}

\exercisequestion{QM-C12-S03-E02}{教材习题 12.3.6：自由粒子的圆周运动}
\textbf{对应知识点：}\relatedtopic{QM-C12-S03-T01}{周期边界与角动量谱}、\relatedtopic{QM-C12-S03-T03}{角向动能与能谱}。

\textbf{题目。}质量 $\mu$ 的无自旋粒子被约束在半径 $a>0$ 的圆周上自由运动，无外加电磁场。证明 $H=L_z^2/(2\mu a^2)$，求能量本征函数、本征值，并解释简并。采用通常周期条件 $\chi(\varphi+2\pi)=\chi(\varphi)$，角向波函数按 $\int_0^{2\pi}|\chi|^2\,\dd\varphi=1$ 归一化。

\begin{checkedsolution}
\textbf{1. 从经典动能得到 Hamiltonian。}切向速度和动能分别为
\begin{equation}
  v=a\dot\varphi,\qquad T=\frac12\mu a^2\dot\varphi^2.
\end{equation}
角坐标的正则动量就是角动量，
\begin{equation}
  p_\varphi=\frac{\partial T}{\partial\dot\varphi}
       =\mu a^2\dot\varphi=l_z,\qquad
  \dot\varphi=\frac{l_z}{\mu a^2}.
\end{equation}
代入 $H=p_\varphi\dot\varphi-T$，得到
\begin{equation}
  H=\frac{l_z^2}{\mu a^2}-\frac{l_z^2}{2\mu a^2}
   =\frac{l_z^2}{2\mu a^2}.
\end{equation}
量子化后，
\begin{equation}
  \boxed{H=\frac{L_z^2}{2\mu a^2}
     =-\frac{\hbar^2}{2\mu a^2}\frac{\dd^2}{\dd\varphi^2}.}
\end{equation}
这是只有角变量的理想圆周模型；不同于平面中心势问题，它没有待求的径向波函数。

\textbf{2. 共同本征函数与能量。}角动量本征函数满足周期条件，故取
\begin{equation}
  \chi_m(\varphi)=\frac{\ee^{\ii m\varphi}}{\sqrt{2\pi}},
  \qquad m\in\mathbb Z,\qquad
  \int_0^{2\pi}|\chi_m|^2\,\dd\varphi=1.
\end{equation}
作用 Hamiltonian，
\begin{equation}
  H\chi_m=\frac{L_z^2}{2\mu a^2}\chi_m
          =\frac{m^2\hbar^2}{2\mu a^2}\chi_m,
  \qquad
  \boxed{E_m=\frac{m^2\hbar^2}{2\mu a^2}.}
\end{equation}

\textbf{3. 简并及物理意义。}令 $E_*=\hbar^2/(2\mu a^2)$，用 $n=|m|=0,1,2,\ldots$ 编号不同能级，则
\begin{equation}
  E_n=n^2E_*,\qquad
  \boxed{g_0=1,\qquad g_n=2\quad(n\geq1).}
\end{equation}

当 $n>0$ 时，$\chi_n$ 和 $\chi_{-n}$ 彼此正交，分别具有角动量 $+n\hbar$ 和 $-n\hbar$，可理解为沿圆周相反方向传播的两个态。能量只依赖 $m^2$，所以两个方向等能；$m=0$ 时两者是同一个态，不能重复计数。

任意归一化组合 $c_+\chi_n+c_-\chi_{-n}$ 仍是该能量的本征态，但一般不再具有确定的 $L_z$。因此二重简并指的是能量本征空间的维数为二，不是该能量下只有两个可能的量子态。
\end{checkedsolution}
