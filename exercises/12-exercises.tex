\exercisequestion{QM-C12-S01-E01}{教材习题 12.1.1：从波函数平移识别生成元}
\textbf{对应知识点：}\relatedtopic{QM-C12-S01-T01}{方向上的动量分量}。题意统一记号重述，缺页及转录来源说明见\hyperref[section:QM-C12-S01]{正文}。

\textbf{题目。}设 $\mathbf n=(n_x,n_y)$ 是固定单位向量，$\psi(x,y)$ 足够光滑。主动平移位移 $\epsilon\mathbf n$ 后，波函数为
\begin{equation}
  \psi_\epsilon(x,y)=\psi(x-\epsilon n_x,y-\epsilon n_y).
\end{equation}
利用 $P_x=-\ii\hbar\partial_x$、$P_y=-\ii\hbar\partial_y$，将一阶展开与
$\psi_\epsilon=(I-\ii\epsilon G_{\mathbf n}/\hbar)\psi+O(\epsilon^2)$ 比较，验证沿 $\mathbf n$ 方向的无穷小平移生成元为 $\mathbf n\cdot\mathbf P$。

\begin{checkedsolution}
对两个自变量同时作一阶 Taylor 展开：
\begin{align}
  \psi_\epsilon(x,y)
  &=\psi(x,y)-\epsilon n_x\partial_x\psi(x,y)
    -\epsilon n_y\partial_y\psi(x,y)+O(\epsilon^2)\notag\\
  &=\left[I-\epsilon(n_x\partial_x+n_y\partial_y)\right]\psi
    +O(\epsilon^2)\notag\\
  &=\left[I-\frac{\ii\epsilon}{\hbar}(n_xP_x+n_yP_y)\right]\psi
    +O(\epsilon^2).
\end{align}
其中 $(-\ii)(-\ii)=-1$ 保证符号一致。对任意上述波函数比较生成元定义，得到
\begin{equation}
  \boxed{G_{\mathbf n}=n_xP_x+n_yP_y=\mathbf n\cdot\mathbf P}.
\end{equation}
\end{checkedsolution}

\exercisequestion{QM-C12-S02-E01}{教材习题 12.2.2：核对角动量的四条对易关系}
\textbf{对应知识点：}\relatedtopic{QM-C12-S02-T02}{分量变换与对易关系}。

\textbf{题目。}教材要求由旋转所需的对易关系确定 $L_z$，或至少验证 $L_z=XP_y-YP_x$ 与之相容。本题采用后一要求。已知
\begin{equation}
  [X,P_x]=[Y,P_y]=\ii\hbar I,\qquad
  [X,Y]=[P_x,P_y]=[X,P_y]=[Y,P_x]=0,
\end{equation}
验证
\begin{equation}
  [X,L_z]=-\ii\hbar Y,\quad [Y,L_z]=\ii\hbar X,\quad
  [P_x,L_z]=-\ii\hbar P_y,\quad [P_y,L_z]=\ii\hbar P_x.
\end{equation}

\begin{checkedsolution}
使用 $[A,BC]=[A,B]C+B[A,C]$。对位置分量，
\begin{align}
  [X,L_z]&=[X,X]P_y+X[X,P_y]-[X,Y]P_x-Y[X,P_x]\notag\\
         &=0+0-0-\ii\hbar Y=\boxed{-\ii\hbar Y},\\
  [Y,L_z]&=[Y,X]P_y+X[Y,P_y]-[Y,Y]P_x-Y[Y,P_x]\notag\\
         &=0+\ii\hbar X-0-0=\boxed{\ii\hbar X}.
\end{align}
对动量分量，注意对易子反对称性给出 $[P_x,X]=[P_y,Y]=-\ii\hbar I$，于是
\begin{align}
  [P_x,L_z]&=[P_x,X]P_y+X[P_x,P_y]-[P_x,Y]P_x-Y[P_x,P_x]\notag\\
           &=-\ii\hbar P_y+0-0-0=\boxed{-\ii\hbar P_y},\\
  [P_y,L_z]&=[P_y,X]P_y+X[P_y,P_y]-[P_y,Y]P_x-Y[P_y,P_x]\notag\\
           &=0+0-(-\ii\hbar P_x)-0=\boxed{\ii\hbar P_x}.
\end{align}
四条关系均成立。将它们用于 $U^\dagger A U=A+\ii\epsilon[L_z,A]/\hbar+O(\epsilon^2)$，位置和动量都按二维无穷小旋转规律变换。

\textbf{严格性补充。}这验证了一致性，却没有单凭对易关系唯一确定生成元：$L_z+cI$（$c$ 为实常数，具有角动量量纲）也满足同样的四条关系。它使旋转算符多出 $\ee^{-\ii\theta c/\hbar}$；正文选用不附加该整体相位的纯轨道旋转约定。
\end{checkedsolution}

\exercisequestion{QM-C12-S02-E02}{教材习题 12.2.3：用两种方法得到极坐标角动量}
\textbf{对应知识点：}\relatedtopic{QM-C12-S02-T02}{有限旋转与极坐标}。

\textbf{题目。}证明 $L_z=-\ii\hbar\partial_\varphi$，其中对 $\varphi$ 的偏导保持 $\rho$ 不变。要求分别使用：
\begin{enumerate}
  \item 坐标变换法：由 $L_z=-\ii\hbar(x\partial_y-y\partial_x)$ 和 $x=\rho\cos\varphi$、$y=\rho\sin\varphi$ 推出结论；
  \item 直接法：将主动小角度旋转 $\psi_\epsilon(\rho,\varphi)=\psi(\rho,\varphi-\epsilon)$ 与生成元定义 $U(\epsilon)=I-\ii\epsilon L_z/\hbar+O(\epsilon^2)$ 比较。
\end{enumerate}

\begin{checkedsolution}
\textbf{方法一：链式法则。}取任意足够光滑的 $f(x,y)$，定义
$F(\rho,\varphi)=f(\rho\cos\varphi,\rho\sin\varphi)$。固定 $\rho$ 时，
\begin{equation}
  \frac{\partial x}{\partial\varphi}=-\rho\sin\varphi=-y,\qquad
  \frac{\partial y}{\partial\varphi}=\rho\cos\varphi=x.
\end{equation}
于是
\begin{align}
  \frac{\partial F}{\partial\varphi}
  &=\frac{\partial f}{\partial x}\frac{\partial x}{\partial\varphi}
    +\frac{\partial f}{\partial y}\frac{\partial y}{\partial\varphi}\notag\\
  &=-y\frac{\partial f}{\partial x}+x\frac{\partial f}{\partial y}
   =(x\partial_y-y\partial_x)f.
\end{align}
因此，同一微分算符在极坐标表示中为
\begin{equation}
  \boxed{L_z=-\ii\hbar(x\partial_y-y\partial_x)
    =-\ii\hbar\partial_\varphi}.
\end{equation}

\textbf{方法二：无穷小旋转。}主动逆时针旋转后，当前极角处的振幅来自原来的 $\varphi-\epsilon$，所以
\begin{align}
  \psi_\epsilon(\rho,\varphi)
  &=\psi(\rho,\varphi-\epsilon)\notag\\
  &=\psi(\rho,\varphi)-\epsilon\partial_\varphi\psi(\rho,\varphi)
    +O(\epsilon^2).
\end{align}
另一方面，生成元定义给出
\begin{equation}
  \psi_\epsilon=\psi-\frac{\ii\epsilon}{\hbar}L_z\psi+O(\epsilon^2).
\end{equation}
对任意上述波函数比较一阶系数，
\begin{equation}
  -\partial_\varphi=-\frac{\ii}{\hbar}L_z
  \quad\Longrightarrow\quad
  \boxed{L_z=-\ii\hbar\partial_\varphi}.
\end{equation}
两种方法分别说明同一算符如何改变坐标表达，以及如何直接由几何变换识别生成元。角导数对应旋转，正如 $P_x=-\ii\hbar\partial_x$ 的位置导数对应平移。
\end{checkedsolution}

\exercisequestion{QM-C12-S03-E01}{教材习题 12.3.3：从角向展开求测量概率}
\textbf{对应知识点：}\relatedtopic{QM-C12-S03-T02}{角向基、径向简并与测量概率}。

\textbf{题目。}平面粒子的归一化波函数为
\begin{equation}
  \psi(\rho,\varphi)=A\ee^{-\rho^2/(2\Delta^2)}\cos^2\varphi,
  \qquad \Delta>0,\quad 0\leq\rho<\infty,\quad 0\leq\varphi<2\pi,
\end{equation}
其中 $A$ 是归一化常数，积分测度为 $\rho\,\dd\rho\,\dd\varphi$。将角向部分展开为 $\Phi_m(\varphi)=\ee^{\ii m\varphi}/\sqrt{2\pi}$，求 $L_z$ 的可能测量结果及概率，并说明为何本题无需显式计算径向积分。

\begin{checkedsolution}
\textbf{1. 识别角向分量。}由倍角公式和 Euler 公式，
\begin{align}
  \cos^2\varphi
  &=\frac{1+\cos2\varphi}{2}
   =\frac12+\frac14\ee^{2\ii\varphi}+\frac14\ee^{-2\ii\varphi}\notag\\
  &=\sqrt{2\pi}\left(\frac12\Phi_0
                   +\frac14\Phi_2+\frac14\Phi_{-2}\right).
\end{align}
只有 $m=0,+2,-2$ 三个分量，故可能测得 $0,+2\hbar,-2\hbar$。复指数前的 $1/2,1/4,1/4$ 不是已归一化的态系数，不能直接将其平方当成最终概率。

\textbf{2. 径向因子为何能约去。}记 $R(\rho)=A\ee^{-\rho^2/(2\Delta^2)}$、$I_R=\int_0^\infty\rho|R|^2\,\dd\rho$，并令
\begin{equation}
  a_0=\frac12,\qquad a_2=a_{-2}=\frac14.
\end{equation}
相对于归一化基 $\Phi_m$，各径向系数为 $f_m=\sqrt{2\pi}\,a_mR$。由角向正交性，
\begin{equation}
  \|\psi\|^2=2\pi I_R\sum_m|a_m|^2,\qquad
  \|\text{第 }m\text{ 个投影分量}\|^2=2\pi I_R|a_m|^2.
\end{equation}
所以归一化后的概率为
\begin{equation}
  P(m)=\frac{2\pi I_R|a_m|^2}
             {2\pi I_R\sum_n|a_n|^2}
      =\frac{|a_m|^2}{\sum_n|a_n|^2}.
\end{equation}
约去的是所有角向分量\emph{共同}的径向积分与基归一化因子，不是径向自由度一般都无关。

\textbf{3. 概率结果。}系数模平方之和为
\begin{equation}
  \sum_m|a_m|^2=\frac14+\frac1{16}+\frac1{16}=\frac38.
\end{equation}
于是
\begin{equation}
  \boxed{P(L_z=0)=\frac23,\qquad
         P(L_z=2\hbar)=\frac16,\qquad
         P(L_z=-2\hbar)=\frac16.}
\end{equation}
三项相加为 $1$；等价的归一化角向态是
$\sqrt{2/3}\,\Phi_0+(\Phi_2+\Phi_{-2})/\sqrt6$。因此本题不必先求出 $A$，结果也不依赖 $\Delta$。
\end{checkedsolution}

\exercisequestion{QM-C12-S03-E02}{教材习题 12.3.6：自由粒子的圆周运动}
\textbf{对应知识点：}\relatedtopic{QM-C12-S03-T01}{周期边界与角动量谱}、\relatedtopic{QM-C12-S03-T03}{角向动能与能谱}。

\textbf{题目。}质量 $\mu$ 的无自旋粒子被约束在半径 $a>0$ 的圆周上自由运动，无外加电磁场。证明 $H=L_z^2/(2\mu a^2)$，求能量本征函数、本征值，并解释简并。采用通常周期条件 $\chi(\varphi+2\pi)=\chi(\varphi)$，角向波函数按 $\int_0^{2\pi}|\chi|^2\,\dd\varphi=1$ 归一化。

\begin{checkedsolution}
\textbf{1. 从经典动能得到 Hamiltonian。}切向速度和动能分别为
\begin{equation}
  v=a\dot\varphi,\qquad T=\frac12\mu a^2\dot\varphi^2.
\end{equation}
角坐标的正则动量就是角动量，
\begin{equation}
  p_\varphi=\frac{\partial T}{\partial\dot\varphi}
       =\mu a^2\dot\varphi=l_z,\qquad
  \dot\varphi=\frac{l_z}{\mu a^2}.
\end{equation}
代入 $H=p_\varphi\dot\varphi-T$，得到
\begin{equation}
  H=\frac{l_z^2}{\mu a^2}-\frac{l_z^2}{2\mu a^2}
   =\frac{l_z^2}{2\mu a^2}.
\end{equation}
量子化后，
\begin{equation}
  \boxed{H=\frac{L_z^2}{2\mu a^2}
     =-\frac{\hbar^2}{2\mu a^2}\frac{\dd^2}{\dd\varphi^2}.}
\end{equation}
这是只有角变量的理想圆周模型；不同于平面中心势问题，它没有待求的径向波函数。

\textbf{2. 共同本征函数与能量。}角动量本征函数满足周期条件，故取
\begin{equation}
  \chi_m(\varphi)=\frac{\ee^{\ii m\varphi}}{\sqrt{2\pi}},
  \qquad m\in\mathbb Z,\qquad
  \int_0^{2\pi}|\chi_m|^2\,\dd\varphi=1.
\end{equation}
作用 Hamiltonian，
\begin{equation}
  H\chi_m=\frac{L_z^2}{2\mu a^2}\chi_m
          =\frac{m^2\hbar^2}{2\mu a^2}\chi_m,
  \qquad
  \boxed{E_m=\frac{m^2\hbar^2}{2\mu a^2}.}
\end{equation}

\textbf{3. 简并及物理意义。}令 $E_*=\hbar^2/(2\mu a^2)$，用 $n=|m|=0,1,2,\ldots$ 编号不同能级，则
\begin{equation}
  E_n=n^2E_*,\qquad
  \boxed{g_0=1,\qquad g_n=2\quad(n\geq1).}
\end{equation}
\begin{center}

  }
  \node[anchor=west] at (4.0,0) {$m=0,\quad g=1$};
  \node[anchor=west] at (4.0,1) {$m=\pm1,\quad g=2$};
  \node[anchor=west] at (4.0,4) {$m=\pm2,\quad g=2$};

\small 圆周自由粒子的低能谱；每个圆点代表一个角动量本征态。

当 $n>0$ 时，$\chi_n$ 和 $\chi_{-n}$ 彼此正交，分别具有角动量 $+n\hbar$ 和 $-n\hbar$，可理解为沿圆周相反方向传播的两个态。能量只依赖 $m^2$，所以两个方向等能；$m=0$ 时两者是同一个态，不能重复计数。

任意归一化组合 $c_+\chi_n+c_-\chi_{-n}$ 仍是该能量的本征态，但一般不再具有确定的 $L_z$。因此二重简并指的是能量本征空间的维数为二，不是该能量下只有两个可能的量子态。
\end{checkedsolution}

\exercisequestion{QM-C12-S04-E01}{教材习题 12.4.3：从三维旋转识别角动量投影}
\textbf{对应知识点：}\relatedtopic{QM-C12-S04-T03}{任意轴旋转与生成元}。以下统一记号重述题意。

\textbf{题目。}无自旋粒子的空间波函数 $\psi(\mathbf r)$ 足够光滑。绕经过原点、沿固定单位向量 $\mathbf n$ 的轴，按右手定则主动旋转小角度 $\epsilon$。
\begin{enumerate}
  \item 证明位置变化为 $\mathbf r'=\mathbf r+\epsilon\mathbf n\times\mathbf r+O(\epsilon^2)$。可先考虑 $\mathbf r\perp\mathbf n$，再推广到一般位置。
  \item 已知 $\psi_\epsilon(\mathbf r)=\psi(R^{-1}\mathbf r)$，其中 $R^{-1}\mathbf r=\mathbf r-\epsilon\mathbf n\times\mathbf r+O(\epsilon^2)$。作一阶 Taylor 展开，与生成元定义
    \begin{equation}
      \psi_\epsilon=\left(I-\frac{\ii\epsilon}{\hbar}G_{\mathbf n}\right)\psi
        +O(\epsilon^2)
    \end{equation}
    比较，利用 $\mathbf P=-\ii\hbar\nabla$、$\mathbf L=\mathbf R\times\mathbf P$，证明 $G_{\mathbf n}=\mathbf n\cdot\mathbf L$。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 先在垂直于旋转轴的平面内证明。}若 $\mathbf r\perp\mathbf n$，则 $\mathbf n\times\mathbf r$ 与 $\mathbf r$ 垂直、长度相同，并指向右手定则规定的正切方向。因此精确的平面旋转为
\begin{equation}
  \mathbf r'=\mathbf r\cos\epsilon
    +(\mathbf n\times\mathbf r)\sin\epsilon.
\end{equation}
使用 $\cos\epsilon=1+O(\epsilon^2)$、$\sin\epsilon=\epsilon+O(\epsilon^3)$，便得
\begin{equation}
  \boxed{\mathbf r'=\mathbf r+\epsilon\mathbf n\times\mathbf r
    +O(\epsilon^2).}
\end{equation}

\begin{center}
\begin{tikzpicture}[>=Stealth,font=\small,scale=0.9]
  \draw[->,thick,noteBlue] (0,0)--(3,0) node[below] {$\mathbf r_\perp$};
  \draw[->,thick,accentRed] (0,0)--(38:3) node[above right] {$\mathbf r'_\perp$};
  \draw[dashed,gray] (0:3) arc[start angle=0,end angle=55,radius=3];
  \draw[->,thick,noteBlue] (3,0)--(3,1.25)
    node[right] {$\epsilon\,\mathbf n\times\mathbf r_\perp$};
  \draw[->] (0:1) arc[start angle=0,end angle=38,radius=1];
  \node at (19:1.35) {$\epsilon$};
  \node[below left] at (0,0) {$O$};
  \node[anchor=west,align=left] at (5,0.7)
    {$\mathbf n$ 垂直纸面向外\\[3pt]平行分量 $\mathbf r_\parallel$ 不变};
\end{tikzpicture}

\small 从轴的正端朝原点看垂直平面；切向箭头只表示一阶变化方向，角度和箭长为示意。
\end{center}

\textbf{2. 推广到一般位置。}将矢量分解为
\begin{equation}
  \mathbf r_\parallel=(\mathbf n\cdot\mathbf r)\mathbf n,\qquad
  \mathbf r_\perp=\mathbf r-\mathbf r_\parallel.
\end{equation}
平行分量在旋转下不变，垂直分量应用上面的结果，故
\begin{align}
  \mathbf r'
  &=\mathbf r_\parallel+\mathbf r_\perp
    +\epsilon\mathbf n\times\mathbf r_\perp+O(\epsilon^2)\notag\\
  &=\boxed{\mathbf r+\epsilon\mathbf n\times\mathbf r+O(\epsilon^2)}.
\end{align}
最后一步使用了 $\mathbf n\times\mathbf r_\parallel=0$。

\textbf{3. 波函数取逆旋转自变量。}主动旋转改变物理状态，而用来描述振幅的坐标点保持固定。新位置 $\mathbf r$ 处的振幅来自旧位置 $R^{-1}\mathbf r$，因此
\begin{align}
  \psi_\epsilon(\mathbf r)
  &=\psi\bigl(\mathbf r-\epsilon\mathbf n\times\mathbf r
      +O(\epsilon^2)\bigr)\notag\\
  &=\psi(\mathbf r)-\epsilon(\mathbf n\times\mathbf r)\cdot
      \nabla\psi(\mathbf r)+O(\epsilon^2)\notag\\
  &=\left[I-\epsilon(\mathbf n\times\mathbf r)\cdot\nabla\right]\psi
      +O(\epsilon^2).
\end{align}
与题目中的生成元定义比较，对任意上述光滑波函数有
\begin{equation}
  -\frac{\ii}{\hbar}G_{\mathbf n}
   =-(\mathbf n\times\mathbf r)\cdot\nabla
  \quad\Longrightarrow\quad
  G_{\mathbf n}=-\ii\hbar(\mathbf n\times\mathbf r)\cdot\nabla.
\end{equation}

\textbf{4. 用标量三重积识别轨道角动量。}在位置表象中
\begin{equation}
  \mathbf L=-\ii\hbar\,\mathbf r\times\nabla.
\end{equation}
对波函数作用时，三重积恒等式给出
\begin{equation}
  (\mathbf n\times\mathbf r)\cdot\nabla\psi
   =\mathbf n\cdot(\mathbf r\times\nabla\psi).
\end{equation}
这里微分作用于右侧的 $\psi$，没有交换位置与微分算符的次序。于是
\begin{equation}
  G_{\mathbf n}\psi
  =-\ii\hbar\,\mathbf n\cdot(\mathbf r\times\nabla\psi)
  =(\mathbf n\cdot\mathbf L)\psi,
  \qquad \boxed{G_{\mathbf n}=\mathbf n\cdot\mathbf L}.
\end{equation}
沿同一轴指数化便得到 $U_{\mathbf n}(\theta)=\exp(-\ii\theta\mathbf n\cdot\mathbf L/\hbar)$。取 $\mathbf n=\mathbf e_z$，恢复
\begin{equation}
  G_z=L_z=-\ii\hbar(x\partial_y-y\partial_x)
      =-\ii\hbar\partial_\varphi,
\end{equation}
所以球坐标中 $\psi_\epsilon(r,\vartheta,\varphi)=\psi(r,\vartheta,\varphi-\epsilon)$，其中 $r,\vartheta$ 保持不变。

\textbf{适用范围。}这里证明的是空间标量波函数的轨道旋转。若还包含自旋自由度，完整旋转生成元需要包含自旋角动量，不能只用 $\mathbf L$ 作用于空间部分来代替。
\end{checkedsolution}

\exercisequestion{QM-C12-S05-E01}{教材习题 12.5.3：横向角动量的涨落}
\textbf{对应知识点：}\relatedtopic{QM-C12-S05-T02}{允许的角动量量子数}、\relatedtopic{QM-C12-S05-T03}{升降系数与矩阵}。

\textbf{题目。}系统处于归一化共同本征态 $\ket{j,m}$，满足
\begin{equation}
  J^2\ket{j,m}=j(j+1)\hbar^2\ket{j,m},\qquad
  J_z\ket{j,m}=m\hbar\ket{j,m}.
\end{equation}
这里 $j=0,\tfrac12,1,\ldots$，$m=-j,-j+1,\ldots,j$，且
\begin{equation}
  J_\pm=J_x\pm\ii J_y,\qquad
  J_\pm\ket{j,m}
   =\hbar\sqrt{(j\mp m)(j\pm m+1)}\ket{j,m\pm1}.
\end{equation}
\begin{enumerate}
  \item 证明 $\expect{J_x}=\expect{J_y}=0$。
  \item 证明 $\expect{J_x^2}=\expect{J_y^2}=\tfrac12\hbar^2[j(j+1)-m^2]$。教材提示可用对称性联系两个二阶矩。
  \item 求 $\Delta J_x\Delta J_y$，验证它满足 Robertson 不确定性关系。
  \item 证明在 $\ket{j,\pm j}$ 中，上述不确定性关系取等号。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 一阶矩。}$J_\pm$ 改变 $m$，所得非零态与原态正交；在阶梯端点处则得到零向量。因此
\begin{equation}
  \expect{J_\pm}=\bra{j,m}J_\pm\ket{j,m}=0.
\end{equation}
利用 $J_x=(J_++J_-)/2$、$J_y=(J_+-J_-)/(2\ii)$，立即有
\begin{equation}
  \boxed{\expect{J_x}=\expect{J_y}=0.}
\end{equation}
这是平均值的陈述，不意味着两个横向分量都确定为零。

\textbf{2. 二阶矩。}绕 $z$ 轴旋转时，$\ket{j,m}$ 只获得整体相位 $\ee^{-\ii m\vartheta}$，所以物理状态不变；旋转 $\pi/2$ 又将 $J_x^2,J_y^2$ 相互交换，故两个期望值相同。结合
\begin{equation}
  J_x^2+J_y^2=J^2-J_z^2
\end{equation}
便得到
\begin{equation}
  \boxed{\expect{J_x^2}=\expect{J_y^2}
    =\frac{\hbar^2}{2}[j(j+1)-m^2].}
\end{equation}

也可按本次作答的升降法直接核对。由于 $J_\pm^2$ 改变 $m$ 两个单位或给出零向量，有 $\expect{J_\pm^2}=0$。保留算符次序：
\begin{align}
  J_x^2&=\frac14(J_+^2+J_+J_-+J_-J_++J_-^2),\notag\\
  J_y^2&=\frac14(-J_+^2+J_+J_-+J_-J_+-J_-^2).
\end{align}
取期望后，
\begin{equation}
  \expect{J_x^2}=\expect{J_y^2}
   =\frac14\expect{J_+J_-+J_-J_+}
   =\frac12\expect{J^2-J_z^2},
\end{equation}
与上面的结果一致。

\textbf{3. 不确定性乘积。}因为一阶矩为零，
\begin{equation}
  (\Delta J_x)^2=(\Delta J_y)^2
    =\frac{\hbar^2}{2}[j(j+1)-m^2],
\end{equation}
所以
\begin{equation}
  \boxed{\Delta J_x\Delta J_y
    =\frac{\hbar^2}{2}[j(j+1)-m^2].}
\end{equation}
而 $[J_x,J_y]=\ii\hbar J_z$ 给出的下界是
\begin{equation}
  \frac12\big|\expect{[J_x,J_y]}\big|
     =\frac{\hbar^2}{2}|m|.
\end{equation}
两边之差非负，因为
\begin{equation}
  j(j+1)-m^2-|m|
    =(j-|m|)(j+|m|+1)\geq0.
\end{equation}
因此
\begin{equation}
  \boxed{\Delta J_x\Delta J_y
     \geq\frac{\hbar^2}{2}|m|
     =\frac{\hbar}{2}|\expect{J_z}|.}
\end{equation}

\textbf{4. 等号条件。}在 $m=\pm j$ 时，
\begin{equation}
  \Delta J_x\Delta J_y
   =\frac{\hbar^2}{2}[j(j+1)-j^2]
   =\frac{\hbar^2j}{2}
   =\frac12\big|\expect{[J_x,J_y]}\big|.
\end{equation}
反过来，由上面的因式分解，在 $\ket{j,m}$ 这一族态中等号成立当且仅当 $|m|=j$。$j=0$ 时两个端点是同一个态。这一“只有端点”的结论限定于所讨论的 $J_z$ 本征态族，不是对任意叠加态作分类。
\end{checkedsolution}

\exercisequestion{QM-C12-S05-E02}{教材习题 12.5.11：构造三个 \texorpdfstring{$l=1$}{l=1} 球谐函数}
\textbf{对应知识点：}\relatedtopic{QM-C12-S05-T05}{球谐函数的最高态与降阶构造}。

\textbf{题目。}由最高态条件
\begin{equation}
  L_zY_1^1=\hbar Y_1^1,\qquad L_+Y_1^1=0
\end{equation}
求出 $Y_1^1(\theta,\varphi)$ 并按球面测度归一化：
\begin{equation}
  \int_0^{2\pi}\dd\varphi\int_0^\pi
    |Y_1^1|^2\sin\theta\,\dd\theta=1.
\end{equation}
最高态采用教材 $(-1)^l$ 的相位约定。然后用 $L_-$ 依次求 $Y_1^0,Y_1^{-1}$，与教材列出的 $l=1$ 球谐函数比较。使用
\begin{equation}
  L_z=-\ii\hbar\partial_\varphi,\qquad
  L_\pm=\hbar\ee^{\pm\ii\varphi}
      (\pm\partial_\theta+\ii\cot\theta\,\partial_\varphi).
\end{equation}

\begin{checkedsolution}
\textbf{1. 最高态的两个角向条件。}由
\begin{equation}
  -\ii\hbar\partial_\varphi Y_1^1=\hbar Y_1^1
\end{equation}
可知 $Y_1^1=f(\theta)\ee^{\ii\varphi}$。再作用 $L_+$：
\begin{align}
  0=L_+Y_1^1
    &=\hbar\ee^{\ii\varphi}
       (\partial_\theta+\ii\cot\theta\,\partial_\varphi)
       [f(\theta)\ee^{\ii\varphi}]\notag\\
    &=\hbar\ee^{2\ii\varphi}[f'(\theta)-f(\theta)\cot\theta].
\end{align}
于是 $f'=f\cot\theta$，解为 $f=C\sin\theta$，即
\begin{equation}
  Y_1^1=C\sin\theta\,\ee^{\ii\varphi}.
\end{equation}

\textbf{2. 归一化与相位。}球面测度不可漏掉 $\sin\theta$：
\begin{equation}
  1=2\pi|C|^2\int_0^\pi\sin^3\theta\,\dd\theta
    =2\pi|C|^2\frac43.
\end{equation}
所以 $|C|^2=3/(8\pi)$。取 $l=1$ 的指定负实系数，
\begin{equation}
  \boxed{Y_1^1=-\sqrt{\frac3{8\pi}}\sin\theta\,\ee^{\ii\varphi}.}
\end{equation}

\textbf{3. 第一次下降。}记 $A=\sqrt{3/(8\pi)}>0$。两个导数分别为
\begin{equation}
  \partial_\theta Y_1^1=-A\cos\theta\,\ee^{\ii\varphi},
  \qquad
  \partial_\varphi Y_1^1=-\ii A\sin\theta\,\ee^{\ii\varphi}.
\end{equation}
因此
\begin{align}
  L_-Y_1^1
   &=-\hbar\ee^{-\ii\varphi}
      \left[-A\cos\theta\,\ee^{\ii\varphi}
      -\ii\cot\theta(-\ii A\sin\theta\,\ee^{\ii\varphi})\right]
      \notag\\
   &=-\hbar\ee^{-\ii\varphi}
      \left[-2A\cos\theta\,\ee^{\ii\varphi}\right]
    =2\hbar A\cos\theta.
\end{align}
括号中第二项也为负，因为 $(-\ii)(-\ii)=-1$。由
$L_-Y_1^1=\hbar\sqrt2\,Y_1^0$，得到
\begin{equation}
  \boxed{Y_1^0=\sqrt2 A\cos\theta
    =\sqrt{\frac3{4\pi}}\cos\theta.}
\end{equation}

\textbf{4. 第二次下降。}此时 $Y_1^0$ 不含 $\varphi$，
\begin{equation}
  L_-Y_1^0
   =-\hbar\ee^{-\ii\varphi}(-\sqrt2 A\sin\theta)
   =\hbar\sqrt2 A\sin\theta\,\ee^{-\ii\varphi}.
\end{equation}
与 $L_-Y_1^0=\hbar\sqrt2\,Y_1^{-1}$ 比较，得到
\begin{equation}
  \boxed{Y_1^{-1}=\sqrt{\frac3{8\pi}}\sin\theta\,\ee^{-\ii\varphi}.}
\end{equation}
三个结果与教材一致，也满足 $Y_1^{-1}=-(Y_1^1)^*$。降阶时已经除以归一化系数，因此不必另给每个结果随意改符号；改变基矢相位时，相关矩阵元和展开系数也必须一致地改变。
\end{checkedsolution}

\exercisequestion{QM-C12-S05-E03}{教材习题 12.5.13：从笛卡尔波函数读出角动量概率}
\textbf{对应知识点：}\relatedtopic{QM-C12-S05-T05}{低阶球谐函数}、\relatedtopic{QM-C12-S05-T06}{径向系数与测量概率}。

\textbf{题目。}无自旋粒子处于已归一化的空间态
\begin{equation}
  \psi(x,y,z)=N(x+y+2z)\ee^{-ar},
  \qquad a>0,\quad r=\sqrt{x^2+y^2+z^2},
\end{equation}
其中 $N$ 为归一化常数。
\begin{enumerate}
  \item 将上一题的球谐函数改写为笛卡尔坐标形式，证明
    \begin{equation}
      Y_1^{\pm1}=\mp\sqrt{\frac3{8\pi}}\frac{x\pm\ii y}{r},
      \qquad Y_1^0=\sqrt{\frac3{4\pi}}\frac zr.
    \end{equation}
  \item 利用这些关系展开 $\psi$，求测量 $L_z$ 的各个可能结果及概率。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 坐标改写。}球坐标给出
\begin{equation}
  \frac{x\pm\ii y}{r}
    =\sin\theta(\cos\varphi\pm\ii\sin\varphi)
    =\sin\theta\,\ee^{\pm\ii\varphi},
  \qquad \frac zr=\cos\theta.
\end{equation}
代入上一题的三个 $Y_1^m$ 即得题目要求的形式。这是球面方向的表达式，$r=0$ 的单点不影响概率积分。

\textbf{2. 反解三个笛卡尔坐标。}令 $A=\sqrt{3/(8\pi)}$，则
\begin{equation}
  Y_1^1=-A\frac{x+\ii y}{r},\qquad
  Y_1^{-1}=A\frac{x-\ii y}{r},\qquad
  Y_1^0=\sqrt2 A\frac zr.
\end{equation}
相加、相减得到
\begin{equation}
  \boxed{\begin{aligned}
    x&=\frac r{2A}(Y_1^{-1}-Y_1^1),\\
    y&=\frac{\ii r}{2A}(Y_1^1+Y_1^{-1}),\\
    z&=\frac r{\sqrt2 A}Y_1^0.
  \end{aligned}}
\end{equation}
所以
\begin{equation}
  x+y+2z=\frac r{2A}
     \left[(-1+\ii)Y_1^1+2\sqrt2Y_1^0+(1+\ii)Y_1^{-1}\right].
\end{equation}
原态变成
\begin{equation}
  \boxed{\psi=\frac{Nr\ee^{-ar}}{2A}
     \left[(-1+\ii)Y_1^1+2\sqrt2Y_1^0+(1+\ii)Y_1^{-1}\right].}
\end{equation}
三个通道都具有 $l=1$，磁量子数为 $+1,0,-1$。它们带有同一径向函数
$F(r)=Nr\ee^{-ar}/(2A)$，但乘着不同复系数。

\textbf{3. 径向积分与归一化概率。}严格的通道概率为
\begin{equation}
  P(m)=\int_0^\infty|C_m(r)|^2r^2\dd r.
\end{equation}
设 $I_r=\int_0^\infty|F(r)|^2r^2\dd r$，则三个概率分别为
\begin{equation}
  |-1+\ii|^2I_r=2I_r,\qquad
  |2\sqrt2|^2I_r=8I_r,\qquad
  |1+\ii|^2I_r=2I_r.
\end{equation}
原态已归一化，且球谐函数彼此正交，所以 $12I_r=1$。因此无需先求 $N$：
\begin{equation}
  \boxed{P(L_z=+\hbar)=\frac16,\qquad
    P(L_z=0)=\frac23,\qquad
    P(L_z=-\hbar)=\frac16.}
\end{equation}
三个概率之和为 $1$。公共径向积分能约去，是本题各角向通道共享同一径向函数的结果，不代表三维问题中可以普遍忽略径向自由度。

另由全部通道都有 $l=1$，可直接读出 $L^2$ 的确定结果为 $2\hbar^2$；$L_z$ 则不确定。原函数在 $\mathbf r\mapsto-\mathbf r$ 下变号，也与 $(-1)^l=-1$ 的奇宇称一致。
\end{checkedsolution}

\exercisequestion{QM-C12-S06-E01}{教材习题 12.6.1：由指数波函数反推势能}
\textbf{对应知识点：}\relatedtopic{QM-C12-S06-T01}{中心势径向方程}、\relatedtopic{QM-C12-S06-T02}{径向边界与渐近行为}。

\textbf{题目。}质量为 $\mu>0$ 的粒子处于
\begin{equation}
  \psi_E(r,\theta,\varphi)=A\ee^{-r/a_0},\qquad a_0>0,
\end{equation}
其中 $A\ne0$ 为归一化常数，不要求计算它。
\begin{enumerate}
  \item 判断该态的角动量内容，即允许的 $l,m$ 及 $L^2,L_z$ 的测量结果。
  \item 假设它是 $H=-\hbar^2\nabla^2/(2\mu)+V(r)$ 的能量本征态，且 $V(r)\to0$ 当 $r\to\infty$，利用无穷远行为求 $E$。
  \item 再考虑有限 $r>0$，求势能 $V(r)$。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 角动量。}波函数无角向依赖，角向部分正比于 $Y_0^0=1/\sqrt{4\pi}$，因此
\begin{equation}
  \boxed{l=m=0,\qquad L^2=0,\qquad L_z=0,}
\end{equation}
两种测量均以概率 $1$ 得到上述结果。这里不是只知道 $m=0$，而是已知完整角向函数为常数。

\textbf{2. 先求导，再取远处极限。}对径向函数使用 $\nabla^2\psi=\psi''+2\psi'/r$：
\begin{equation}
  \psi_E'=-\frac{\psi_E}{a_0},\qquad
  \psi_E''=\frac{\psi_E}{a_0^2},\qquad
  \nabla^2\psi_E=\left(\frac1{a_0^2}-\frac2{a_0r}\right)\psi_E.
\end{equation}
在有限 $r>0$ 先除以非零的 $\psi_E$，得到
\begin{equation}
  E=-\frac{\hbar^2}{2\mu a_0^2}+\frac{\hbar^2}{\mu a_0r}+V(r).
\end{equation}
随后令 $r\to\infty$，利用 $V(\infty)=0$，有
\begin{equation}
  \boxed{E=-\frac{\hbar^2}{2\mu a_0^2}.}
\end{equation}
若先让波函数本身趋零，容易只得到无信息的 $0=0$。能量为负是相对于所指定的远处势能零点而言的。

\textbf{3. 反求势能。}将 $E$ 代回有限 $r$ 的等式，常数项抵消：
\begin{equation}
  \boxed{V(r)=-\frac{\hbar^2}{\mu a_0r},\qquad r>0.}
\end{equation}
它是吸引型库仑势 $-g/r$，其中 $g=\hbar^2/(\mu a_0)$ 的量纲为能量乘长度。已知态的角向结构、远处衰减与完整微分方程在这里分别确定角动量、能量和势能。
\end{checkedsolution}

\exercisequestion{QM-C12-S06-E02}{教材习题 12.6.9：球形有限阱的束缚阈值}
\textbf{对应知识点：}\relatedtopic{QM-C12-S06-T01}{半轴径向方程}、\relatedtopic{QM-C12-S06-T02}{边界与可归一化条件}。

\textbf{题目。}质量为 $\mu$ 的粒子处于
\begin{equation}
  V(r)=\begin{cases}-V_0,&0\leq r<r_0,\\0,&r>r_0,\end{cases}
  \qquad V_0>0,\quad r_0>0.
\end{equation}
只研究 $l=0$，写 $\psi=u(r)Y_0^0/r$。对束缚能量 $-V_0<E<0$，定义
\begin{equation}
  k=\frac{\sqrt{2\mu(E+V_0)}}\hbar,\qquad
  \kappa=\frac{\sqrt{-2\mu E}}\hbar.
\end{equation}
这里的 $k$ 对应教材题干的 $k'$。
\begin{enumerate}
  \item 求阱内外满足原点条件与远处衰减的 $u(r)$，暂不归一化。
  \item 在 $r=r_0$ 匹配，证明 $k\cot(kr_0)=-\kappa$，即 $k/\kappa=-\tan(kr_0)$。
  \item 证明 $V_0<\pi^2\hbar^2/(8\mu r_0^2)$ 时无 $l=0$ 束缚态；补充检查等号时的 $E=0$ 阈值解是否平方可积。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 两端边界先筛选解。}阱内有 $u''+k^2u=0$；通解 $A\sin kr+B\cos kr$ 经 $u(0)=0$ 筛选后只剩正弦。阱外有 $u''-\kappa^2u=0$，丢弃远处增长支：
\begin{equation}
  \boxed{u(r)=\begin{cases}
    A\sin(kr),&0\leq r<r_0,\\
    C\ee^{-\kappa r},&r>r_0.
  \end{cases}}
\end{equation}
原点排除余弦支是通常三维 Hamiltonian 的边界要求，不能仅以 $1/r$ 奇异性局部不可积为理由，因为它实际局部平方可积。

\textbf{2. 边界匹配。}有限势阶跃不含 delta 项，所以 $u,u'$ 连续：
\begin{equation}
  A\sin(kr_0)=C\ee^{-\kappa r_0},\qquad
  Ak\cos(kr_0)=-\kappa C\ee^{-\kappa r_0}.
\end{equation}
非平凡解不能有 $\sin(kr_0)=0$：否则第一式令 $C=0$，第二式再令 $A=0$。故可以相除，得到
\begin{equation}
  \boxed{k\cot(kr_0)=-\kappa,\qquad
    \frac{k}{\kappa}=-\tan(kr_0).}
\end{equation}
第二种写法只在 $\kappa>0$ 时使用，不能在阈值处直接除以零。

\textbf{3. 阱深的必要条件。}记
\begin{equation}
  z=kr_0,\qquad \beta=\kappa r_0,\qquad
  z_0=\frac{r_0\sqrt{2\mu V_0}}\hbar.
\end{equation}
由定义，
\begin{equation}
  z^2+\beta^2=z_0^2,\qquad z\cot z=-\beta,
  \qquad z>0,\quad\beta>0.
\end{equation}
若 $z_0\leq\pi/2$，则 $0<z<z_0\leq\pi/2$，左边 $z\cot z>0$，右边 $-\beta<0$，无法相等。因此
\begin{equation}
  \boxed{V_0\leq V_{0,c}=\frac{\pi^2\hbar^2}{8\mu r_0^2}
    \quad\Longrightarrow\quad\text{无负能量的 }l=0\text{ 束缚态}.}
\end{equation}

\textbf{4. 临界等号的零能解。}当 $V_0=V_{0,c}$、$E=0$ 时，$k=\pi/(2r_0)$。阱外须重新解 $u''=0$，通解是 $Cr+D$。阱内解在边界满足 $u(r_0)=A$、$u'(r_0)=0$，匹配给出 $C=0,D=A$：
\begin{equation}
  \boxed{u(r)=\begin{cases}
    A\sin\bigl(\pi r/(2r_0)\bigr),&0\leq r<r_0,\\
    A,&r\geq r_0.
  \end{cases}}
\end{equation}

\begin{center}
\begin{tikzpicture}[x=2.1cm,y=1.7cm,>=Stealth]
  \draw[->] (0,0)--(3.65,0) node[right] {$r/r_0$};
  \draw[->] (0,0)--(0,1.5) node[above] {$u/A$};
  \draw[dashed,gray] (1,0)--(1,1.22);
  \draw[thick,noteBlue,domain=0:1,samples=60] plot (\x,{sin(90*\x)});
  \draw[thick,noteBlue,->] (1,1)--(3.5,1);
  \node[below] at (1,0) {$1$};
  \node[left] at (0,1) {$1$};
  \node at (0.52,-0.4) {阱内};
  \node at (2.4,-0.4) {阱外};
  \node[above] at (2.3,1.04) {$u=A,\quad\psi\propto1/r$};
\end{tikzpicture}

\small 临界解示意：纵轴是 $u/A$，不是三维波函数或概率密度。
\end{center}

由于 $\int|Y_0^0|^2\dd\Omega=1$，
\begin{equation}
  \int|\psi|^2\dd^3r=\int_0^\infty|u|^2\dd r,
  \qquad \int_{r_0}^\infty|u|^2\dd r
     =|A|^2\int_{r_0}^\infty\dd r=\infty.
\end{equation}
所以非零阈值解不可归一化。虽然 $\psi\sim1/r\to0$，仍不足以克服体积元 $r^2\dd r$；\textbf{趋零不是平方可积的充分条件}。

\dialoguesupplement{若要把阈值写成充分必要条件，还须补上存在性。$z_0>\pi/2$ 时，在 $\pi/2<z<\min(z_0,\pi)$ 考察 $F(z)=-z\cot z-\sqrt{z_0^2-z^2}$。它在左端趋于负值，在右端趋于正值或正无穷，故连续性保证内部有根，且 $\beta>0$。因此至少一个真正的 $l=0$ 束缚态存在当且仅当 $V_0>V_{0,c}$。}

三维球形阱的这个有限阈值与一维全实轴吸引阱不同：径向半轴必须满足 $u(0)=0$，等价于对称一维阱的奇态边界，而不是可在原点取非零值的偶态边界。
\end{checkedsolution}

\exercisequestion{QM-C12-S06-E03}{教材习题 12.6.11 第 (2) 问：简并度与宇称}
\textbf{对应知识点：}\relatedtopic{QM-C12-S06-T05}{三维各向同性谐振子}、\relatedtopic{QM-C12-S05-T06}{球谐函数的空间宇称}。

\textbf{题目。}不计自旋的三维各向同性谐振子满足
\begin{equation}
  E_N=\left(N+\frac32\right)\hbar\omega,\qquad
  N=2n_r+l,\qquad l=N,N-2,\ldots,0\text{ 或 }1.
\end{equation}
每个 $l$ 下 $m=-l,\ldots,l$。计算每个能级的简并度与宇称，并与笛卡尔坐标解比较。具体分为：
\begin{enumerate}
  \item 求 $g_N=\sum_{l=N,N-2,\ldots}(2l+1)$ 的闭式。
  \item 由 $\Pi[R(r)Y_l^m]=(-1)^lR(r)Y_l^m$，判断同一能级的宇称；检查其中任意非零线性组合是否仍有确定宇称。
  \item 独立统计 $n_x+n_y+n_z=N$ 的有序非负整数三元组，比较两种基底下的计数。
\end{enumerate}
教材要求与 10.2.3 的笛卡尔结果比较；该旧题未曾独立选做，这里直接完成所需计数，不追记为整道旧题已完成。

\begin{checkedsolution}
\textbf{1. 分偶数、奇数求和。}若 $N=2p$，则 $l=0,2,\ldots,2p$，
\begin{equation}
  g_{2p}=\sum_{j=0}^p(4j+1)
    =2p(p+1)+(p+1)=(p+1)(2p+1).
\end{equation}
若 $N=2p+1$，则 $l=1,3,\ldots,2p+1$，
\begin{equation}
  g_{2p+1}=\sum_{j=0}^p(4j+3)
    =2p(p+1)+3(p+1)=(p+1)(2p+3).
\end{equation}
两种情况统一为
\begin{equation}
  \boxed{g_N=\frac{(N+1)(N+2)}2.}
\end{equation}
这里计数的是独立空间本征态，不包含自旋多重度。

\textbf{2. 同一能级的宇称相同。}由 $l=N-2n_r$，
\begin{equation}
  (-1)^l=(-1)^{N-2n_r}=(-1)^N,
  \qquad \boxed{\pi_N=(-1)^N.}
\end{equation}
所以固定能级中的任意非零态
\begin{equation}
  \ket\Psi=\sum_{l,m}c_{lm}\ket{N,l,m}
\end{equation}
均满足
\begin{equation}
  \Pi\ket\Psi=\sum_{l,m}c_{lm}(-1)^l\ket{N,l,m}
     =(-1)^N\ket\Psi.
\end{equation}
因此即使混合不同 $l$，仍是同一宇称本征态；但一般不再是 $L^2$ 本征态。若混合不同 $m$，也一般不是 $L_z$ 本征态。这个“同能级同宇称”的结论来自本系统允许的 $l$ 结构，不能推广为所有 Hamiltonian 的普遍规律。

\textbf{3. 笛卡尔计数。}将 $N$ 个相同激发量子分配到三个有标签的方向，允许某个方向分得零个。用两个隔板分出三组，故
\begin{equation}
  g_N=\binom{N+2}{2}=\boxed{\frac{(N+1)(N+2)}2}.
\end{equation}
两种计数相同，因为它们是在同一个能量本征空间中选择不同基底，而不是得到不同的物理空间。笛卡尔乘积态的宇称也为 $(-1)^{n_x+n_y+n_z}=(-1)^N$。
