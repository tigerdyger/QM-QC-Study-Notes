\exercisequestion{QM-C14-S03-E01}{教材正文示范：一次旋转得到任意方向的自旋态}
\textbf{来源：}印刷 pp.~383--384，式 (14.3.45)--(14.3.47)，无独立 Example 编号。

\textbf{对应知识点：}\relatedtopic{QM-C14-S03-T05}{有限旋转与半角}。

\textbf{示范任务。}初态为 $\ket{z,+}=(1,0)^{\mathsf T}$。目标方向为
$\hat{\mathbf n}=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$。
先取 $0<\theta<\pi$，选择把 $+\hat{\mathbf z}$ 转到 $\hat{\mathbf n}$ 的旋转轴，用
\begin{equation}
  U_{\mathbf a}(\alpha)=\cos\frac\alpha2 I
    -\ii\sin\frac\alpha2\hat{\mathbf a}\cdot\boldsymbol\sigma
\end{equation}
求旋转后的态，并与教材相位约定
$\ket{\hat{\mathbf n},+}=(\cos(\theta/2)\ee^{-\ii\varphi/2},
\sin(\theta/2)\ee^{\ii\varphi/2})^{\mathsf T}$ 比较。

\begin{checkedsolution}
按右手定则选轴和角度：
\begin{equation}
  \hat{\mathbf a}=\frac{\hat{\mathbf z}\times\hat{\mathbf n}}
     {|\hat{\mathbf z}\times\hat{\mathbf n}|}
     =(-\sin\varphi,\cos\varphi,0),\qquad \alpha=\theta.
\end{equation}
由 Pauli 矩阵，
\begin{equation}
  \hat{\mathbf a}\cdot\boldsymbol\sigma
    =\begin{pmatrix}0&-\ii\ee^{-\ii\varphi}\\
      \ii\ee^{\ii\varphi}&0\end{pmatrix},\qquad
  U_{\mathbf a}(\theta)=\begin{pmatrix}
      \cos(\theta/2)&-\sin(\theta/2)\ee^{-\ii\varphi}\\
      \sin(\theta/2)\ee^{\ii\varphi}&\cos(\theta/2)
    \end{pmatrix}.
\end{equation}
作用于初态即取第一列：
\begin{equation}
  \boxed{U_{\mathbf a}(\theta)\ket{z,+}
    =\begin{pmatrix}\cos(\theta/2)\\\sin(\theta/2)\ee^{\ii\varphi}\end{pmatrix}
    =\ee^{\ii\varphi/2}\ket{\hat{\mathbf n},+}.}
\end{equation}
两者只差整体相位，物理状态相同。$\theta=0,\pi$ 时不能再用叉积归一化来定义轴，但上面的显式选轴与旋转矩阵仍适用；极点处方位角不再指定不同的目标方向。
\end{checkedsolution}

\exercisequestion{QM-C14-S03-E02}{教材习题 14.3.2：本征旋量与自旋期望值}
\textbf{对应知识点：}\relatedtopic{QM-C14-S03-T03}{任意方向的自旋态}。

\textbf{题目。}在 $S_z$ 本征基下取
\begin{equation}
  S_x=\frac\hbar2\begin{pmatrix}0&1\\1&0\end{pmatrix},\quad
  S_y=\frac\hbar2\begin{pmatrix}0&-\ii\\\ii&0\end{pmatrix},\quad
  S_z=\frac\hbar2\begin{pmatrix}1&0\\0&-1\end{pmatrix}.
\end{equation}
令 $\hat{\mathbf n}=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$，$0\le\theta\le\pi$，并给出
\begin{equation}
  \chi_+=\begin{pmatrix}\cos(\theta/2)\ee^{-\ii\varphi/2}\\
      \sin(\theta/2)\ee^{\ii\varphi/2}\end{pmatrix},\qquad
  \chi_-=\begin{pmatrix}-\sin(\theta/2)\ee^{-\ii\varphi/2}\\
      \cos(\theta/2)\ee^{\ii\varphi/2}\end{pmatrix}.
\end{equation}
验证 $S_{\mathbf n}\chi_\pm=\pm\hbar\chi_\pm/2$，并分别计算三个分量的期望值，证明 $\expect{\mathbf S}_\pm=\pm\hbar\hat{\mathbf n}/2$。原题使用 $\hat{\mathbf n}\cdot\boldsymbol\sigma$ 的本征值 $\pm1$，与此完全等价。

\begin{checkedsolution}
\textbf{1. 本征态。}记 $c=\cos(\theta/2)$、$s=\sin(\theta/2)$。由
$\cos\theta=c^2-s^2$、$\sin\theta=2cs$，可得
\begin{equation}
  c\cos\theta+s\sin\theta=c,\qquad
  c\sin\theta-s\cos\theta=s.
\end{equation}
于是直接相乘给出
\begin{align}
  S_{\mathbf n}\chi_+
   &=\frac\hbar2\begin{pmatrix}
     \ee^{-\ii\varphi/2}(c\cos\theta+s\sin\theta)\\
     \ee^{\ii\varphi/2}(c\sin\theta-s\cos\theta)
     \end{pmatrix}=\frac\hbar2\chi_+,\\
  S_{\mathbf n}\chi_-
   &=\frac\hbar2\begin{pmatrix}
     \ee^{-\ii\varphi/2}(-s\cos\theta+c\sin\theta)\\
     \ee^{\ii\varphi/2}(-s\sin\theta-c\cos\theta)
     \end{pmatrix}=-\frac\hbar2\chi_-.
\end{align}
同时 $\chi_\pm^\dagger\chi_\pm=c^2+s^2=1$、$\chi_+^\dagger\chi_-=0$，故确为正交归一本征基。

\textbf{2. 期望值。}对任意归一化列向量 $\chi=(a,b)^{\mathsf T}$，直接矩阵相乘得到
\begin{equation}
  \expect{S_x}=\hbar\operatorname{Re}(a^*b),\qquad
  \expect{S_y}=\hbar\operatorname{Im}(a^*b),\qquad
  \expect{S_z}=\frac\hbar2(|a|^2-|b|^2).
\end{equation}
其中 $S_y$ 的组合是 $\hbar(-\ii a^*b+\ii b^*a)/2$，取伴随时相位须共轭。对 $\chi_+$ 有 $a^*b=cs\ee^{\ii\varphi}$，对 $\chi_-$ 有 $a^*b=-cs\ee^{\ii\varphi}$，所以
\begin{equation}
  \begin{array}{c|ccc}
    &\expect{S_x}&\expect{S_y}&\expect{S_z}\\ \hline
    \chi_+&\hbar cs\cos\varphi&\hbar cs\sin\varphi&\hbar(c^2-s^2)/2\\
    \chi_-&-\hbar cs\cos\varphi&-\hbar cs\sin\varphi&\hbar(s^2-c^2)/2
  \end{array}
\end{equation}
利用 $2cs=\sin\theta$，最终
\begin{equation}
  \boxed{\expect{\mathbf S}_\pm
    =\pm\frac\hbar2(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta).}
\end{equation}
这些是平均值，不表示三个分量同时具有确定测量值。
\end{checkedsolution}

\exercisequestion{QM-C14-S03-E03}{教材习题 14.3.4：由对易关系推导乘积恒等式}
\textbf{对应知识点：}\relatedtopic{QM-C14-S03-T04}{Pauli 矩阵代数}。

\textbf{题目。}只采用原题的第一种方法。设 $\mathbf A,\mathbf B$ 是普通实数矢量，已知
\begin{equation}
  [\sigma_i,\sigma_j]=2\ii\sum_k\epsilon_{ijk}\sigma_k,
  \qquad \{\sigma_i,\sigma_j\}=2\delta_{ij}I.
\end{equation}
先将 $\sigma_i\sigma_j$ 用对易子与反对易子表示，再证明
$(\mathbf A\cdot\boldsymbol\sigma)(\mathbf B\cdot\boldsymbol\sigma)
=(\mathbf A\cdot\mathbf B)I+\ii(\mathbf A\times\mathbf B)\cdot\boldsymbol\sigma$。

\begin{checkedsolution}
将对易子与反对易子相加：
\begin{equation}
  \sigma_i\sigma_j=\frac12\bigl(\{\sigma_i,\sigma_j\}+[\sigma_i,\sigma_j]\bigr)
    =\delta_{ij}I+\ii\sum_k\epsilon_{ijk}\sigma_k.
\end{equation}
于是
\begin{align}
  (\mathbf A\cdot\boldsymbol\sigma)(\mathbf B\cdot\boldsymbol\sigma)
    &=\sum_{i,j}A_iB_j\sigma_i\sigma_j\notag\\
    &=\sum_iA_iB_iI
       +\ii\sum_k\left(\sum_{i,j}\epsilon_{ijk}A_iB_j\right)\sigma_k.
\end{align}
第一项是 $(\mathbf A\cdot\mathbf B)I$。循环置换不改变 Levi-Civita 符号，故
\begin{equation}
  \epsilon_{ijk}=\epsilon_{kij},\qquad
  \sum_{i,j}\epsilon_{ijk}A_iB_j=(\mathbf A\times\mathbf B)_k.
\end{equation}
因此
\begin{equation}
  \boxed{(\mathbf A\cdot\boldsymbol\sigma)(\mathbf B\cdot\boldsymbol\sigma)
    =(\mathbf A\cdot\mathbf B)I
     +\ii(\mathbf A\times\mathbf B)\cdot\boldsymbol\sigma.}
\end{equation}
这里是循环置换，不是交换两个指标，所以没有额外负号；标量点积项前的单位矩阵也不可省略其含义。
\end{checkedsolution}

\exercisequestion{QM-C14-S03-E04}{教材习题 14.3.6：先绕固定 y 轴，再绕固定 z 轴}
\textbf{对应知识点：}\relatedtopic{QM-C14-S03-T05}{有限旋转与顺序}。

\textbf{题目。}初态为 $\ket{z,+}=(1,0)^{\mathsf T}$，目标方向为
$\hat{\mathbf n}=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$。
采用右手定则的主动旋转，且两次均绕固定坐标轴。给定
\begin{equation}
  U_y(\theta)=\ee^{-\ii\theta\sigma_y/2},\quad
  U_z(\varphi)=\ee^{-\ii\varphi\sigma_z/2},\quad
  \sigma_y=\begin{pmatrix}0&-\ii\\\ii&0\end{pmatrix},\quad
  \sigma_z=\begin{pmatrix}1&0\\0&-1\end{pmatrix}.
\end{equation}
先从几何说明 $R_z(\varphi)R_y(\theta)\hat{\mathbf z}=\hat{\mathbf n}$ 及算符的作用顺序，再用 $\ee^{-\ii\alpha\sigma_i/2}=\cos(\alpha/2)I-\ii\sin(\alpha/2)\sigma_i$ 显式验证
\begin{equation}
  U_z(\varphi)U_y(\theta)\ket{z,+}
    =\begin{pmatrix}\cos(\theta/2)\ee^{-\ii\varphi/2}\\
      \sin(\theta/2)\ee^{\ii\varphi/2}\end{pmatrix}
    =\ket{\hat{\mathbf n},+}.
\end{equation}

\begin{checkedsolution}
\textbf{1. 几何与顺序。}先绕 $y$ 轴改变极角，再绕 $z$ 轴改变方位角：
\begin{equation}
  \begin{pmatrix}0\\0\\1\end{pmatrix}
    \xrightarrow{R_y(\theta)}
  \begin{pmatrix}\sin\theta\\0\\\cos\theta\end{pmatrix}
    \xrightarrow{R_z(\varphi)}
  \begin{pmatrix}\sin\theta\cos\varphi\\\sin\theta\sin\varphi\\\cos\theta\end{pmatrix}.
\end{equation}
\begin{center}
\begin{tikzpicture}[>=Stealth,scale=0.9,font=\small]
  \begin{scope}
    \draw[->] (0,0)--(2.55,0) node[right] {$x$};
    \draw[->] (0,0)--(0,2.5) node[above] {$z$};
    \draw[->,very thick,noteBlue] (0,0)--(0,2.1);
    \draw[->,very thick,accentRed] (0,0)--(40:2.1)
      node[above right] {$\mathbf n_1$};
    \draw[->] (0,0.72) arc[start angle=90,end angle=40,radius=0.72];
    \node at (0.39,0.9) {$\theta$};
    \node[align=center] at (1.05,-0.6) {先绕 $y$ 轴\\$xz$ 平面};
  \end{scope}
  \begin{scope}[xshift=6cm]
    \draw[->] (0,0)--(2.55,0) node[right] {$x$};
    \draw[->] (0,0)--(0,2.5) node[above] {$y$};
    \draw[->,very thick,noteBlue] (0,0)--(2.1,0);
    \draw[->,very thick,accentRed] (0,0)--(40:2.1)
      node[above right] {$\mathbf n_{xy}$};
    \draw[->] (0.72,0) arc[start angle=0,end angle=40,radius=0.72];
    \node at (0.98,0.35) {$\varphi$};
    \node[align=center] at (1.05,-0.6) {再绕 $z$ 轴\\仅画 $xy$ 投影};
  \end{scope}
\end{tikzpicture}

{\small 原创方向示意，角度与长度不按同一比例；第二步保持 $z$ 分量 $\cos\theta$ 不变。}
\end{center}
列向量从右向左受作用，所以先发生的 $U_y$ 写在右边，后发生的 $U_z$ 写在左边。几何只能确定最终自旋方向，即物理态的射线；具体整体相位须由下面的矩阵计算确定。

\textbf{2. 显式计算。}记 $c=\cos(\theta/2)$、$s=\sin(\theta/2)$。由 $(-\ii)(-\ii)=-1$、$(-\ii)\ii=1$，
\begin{equation}
  U_y(\theta)=\begin{pmatrix}c&-s\\s&c\end{pmatrix},\qquad
  U_z(\varphi)=\begin{pmatrix}\ee^{-\ii\varphi/2}&0\\0&\ee^{\ii\varphi/2}\end{pmatrix}.
\end{equation}
依次作用得到
\begin{equation}
  U_y(\theta)\ket{z,+}=\begin{pmatrix}c\\s\end{pmatrix},\qquad
  \boxed{U_z(\varphi)U_y(\theta)\ket{z,+}
    =\begin{pmatrix}c\ee^{-\ii\varphi/2}\\s\ee^{\ii\varphi/2}\end{pmatrix}
    =\ket{\hat{\mathbf n},+}.}
\end{equation}
在本题约定下是态矢量的精确相等；与前面的单轴旋转示范相比，不同的旋转操作将此初态送到同一方向，但所得代表矢量可以相差整体相位。
\end{checkedsolution}

\exercisequestion{QM-C14-S04-E01}{教材习题 14.4.1：期望值的精确进动方程}
\textbf{对应知识点：}\relatedtopic{QM-C14-S04-T01}{磁矩与进动}、\relatedtopic{QM-C14-S04-T03}{自旋时间演化}。

\textbf{题目。}给定 $H=-\gamma\mathbf L\cdot\mathbf B$、$\boldsymbol\mu=\gamma\mathbf L$，其中 $\gamma$ 为常数，$\mathbf B$ 是不依赖位置的经典外场。证明
\begin{equation}
  \frac{\dd\expect{\mathbf L}}{\dd t}
   =\expect{\boldsymbol\mu\times\mathbf B}
   =\expect{\boldsymbol\mu}\times\mathbf B.
\end{equation}
并说明教材随后指出的两种推广：$\mathbf B$ 可以依赖时间；将 $\mathbf L$ 换成 $\mathbf S$ 后同样成立。可用 $[L_i,L_j]=\ii\hbar\sum_k\epsilon_{ijk}L_k$ 与 Ehrenfest 定理。

\begin{checkedsolution}
\textbf{1. 一个分量。}外场分量是与 $\mathbf L$ 对易的数，故
\begin{align}
  [L_x,H]
    &=-\gamma\bigl(B_x[L_x,L_x]+B_y[L_x,L_y]+B_z[L_x,L_z]\bigr)\notag\\
    &=-\gamma\bigl(\ii\hbar B_yL_z-\ii\hbar B_zL_y\bigr)
      =\ii\hbar\gamma(L_yB_z-L_zB_y).
\end{align}
$L_x$ 不显含时间，因此
\begin{equation}
  \frac{\dd\expect{L_x}}{\dd t}
    =\frac1{\ii\hbar}\expect{[L_x,H]}
    =\gamma\expect{L_yB_z-L_zB_y}
    =\expect{(\boldsymbol\mu\times\mathbf B)_x}.
\end{equation}
循环置换 $x\to y\to z\to x$ 得到另两个分量，合并并把外场提出期望值：
\begin{equation}
  \boxed{\frac{\dd\expect{\mathbf L}}{\dd t}
     =\expect{\boldsymbol\mu\times\mathbf B}
     =\expect{\boldsymbol\mu}\times\mathbf B.}
\end{equation}

\textbf{2. 推广。}若 $\mathbf B=\mathbf B(t)$，上述每一时刻的对易计算不变；$H$ 含时并不在 $\dd\expect{\mathbf L}/\dd t$ 中额外产生 $\expect{\partial_tH}$，显含时间项应检查的是可观测量 $\mathbf L$。因此
\begin{equation}
  \frac{\dd\expect{\mathbf L}}{\dd t}
       =\gamma\expect{\mathbf L}\times\mathbf B(t).
\end{equation}
此时有效轴可变，不必是固定轴匀速进动。自旋满足同样的角动量代数，故在 $H_S=-\gamma\mathbf S\cdot\mathbf B(t)$ 下，
\begin{equation}
  \boxed{\frac{\dd\expect{\mathbf S}}{\dd t}
       =\gamma\expect{\mathbf S}\times\mathbf B(t).}
\end{equation}
应取该粒子自旋磁矩对应的 $\gamma$，不强令它等于轨道旋磁比。结论对给定 Hamiltonian 精确成立，不要求经典极限；也不意味着量子态具有同时确定的三个角动量分量。
\end{checkedsolution}

\exercisequestion{QM-C14-S04-E02}{教材习题 14.4.3：旋转系中的精确求解}
\textbf{对应知识点：}\relatedtopic{QM-C14-S04-T04}{旋转磁场、共振与脉冲}。

\textbf{题目与约定。}电子自旋满足 $H=-\gamma\mathbf S\cdot\mathbf B(t)$、$\gamma<0$，初态为 $\ket{\psi(0)}=\ket{z,+}=(1,0)^{\mathsf T}$。采用本笔记的正向旋转场
\begin{equation}
  \mathbf B(t)=B_0\hat{\mathbf z}
    +B_1(\cos\omega t\,\hat{\mathbf x}+\sin\omega t\,\hat{\mathbf y}),
  \qquad B_0,B_1>0,
\end{equation}
其中驱动频率和幅度恒定。教材用 $B\cos\omega_{\rm book}t\,\hat{\mathbf x}-B\sin\omega_{\rm book}t\,\hat{\mathbf y}$；取 $B=B_1$、$\omega_{\rm book}=-\omega$ 即与这里相同，态变换也随之转换。定义
\begin{align}
  &\Omega_0=-\gamma B_0,\quad \Omega_1=-\gamma B_1,\quad
  \Delta=\Omega_0-\omega,\quad \Omega_R=\sqrt{\Delta^2+\Omega_1^2},\\
  &\ket{\psi_r}=V\ket\psi,\qquad V=\ee^{+\ii\omega tS_z/\hbar}.
\end{align}
\begin{enumerate}
  \item 从实验室 Schrödinger 方程推导旋转系方程，求不含时间的 $H_r$。
  \item 求 $U_r(t)$，显式写出旋转系及实验室系中的两分量态，并检查初值与归一化。
  \item 比较共振时的态与沿某方向的自旋纯态，计算 $\expect{\mu_z}_t$，验证教材式 (14.4.31) 的进动几何结构。由同一结果读出翻转概率、第一次完全翻转的时间及固定失谐时的最大概率。
\end{enumerate}

\begin{checkedsolution}
\textbf{1. 含时变换。}乘积求导并用 $\ket\psi=V^\dagger\ket{\psi_r}$：
\begin{align}
  \ii\hbar\frac{\dd}{\dd t}\ket{\psi_r}
    &=\ii\hbar\dot V\ket\psi+VH\ket\psi\notag\\
    &=\left(VHV^\dagger+\ii\hbar\dot VV^\dagger\right)\ket{\psi_r}.
\end{align}
在 $S_z$ 基底下，
\begin{equation}
  V=\begin{pmatrix}\ee^{\ii\omega t/2}&0\\0&\ee^{-\ii\omega t/2}\end{pmatrix},
  \qquad H(t)=\frac\hbar2
    \begin{pmatrix}\Omega_0&\Omega_1\ee^{-\ii\omega t}\\
      \Omega_1\ee^{\ii\omega t}&-\Omega_0\end{pmatrix}.
\end{equation}
左右相乘消去非对角元的时间相位，且 $\dot V=(\ii\omega S_z/\hbar)V$，所以
\begin{equation}
  VHV^\dagger=\Omega_0S_z+\Omega_1S_x,\qquad
  \ii\hbar\dot VV^\dagger=-\omega S_z,
  \qquad \boxed{H_r=\Delta S_z+\Omega_1S_x.}
\end{equation}
后一个负号来自 $\ii^2=-1$，不能把含时变换仅当成相似变换。

\textbf{2. 求矩阵指数。}对 $\Omega_R>0$，令
\begin{equation}
  M=\frac{\Delta\sigma_z+\Omega_1\sigma_x}{\Omega_R}.
\end{equation}
由 $\sigma_x^2=\sigma_z^2=I$、$\{\sigma_x,\sigma_z\}=0$，
\begin{equation}
  M^2=\frac{\Delta^2+\Omega_1^2}{\Omega_R^2}I=I.
\end{equation}
令 $c_t=\cos(\Omega_Rt/2)$、$s_t=\sin(\Omega_Rt/2)$。指数级数的偶次幂给出 $c_tI$，奇次幂给出 $-\ii s_tM$，因此
\begin{equation}
  \boxed{U_r(t)=c_tI-\ii s_tM
   =\begin{pmatrix}
     c_t-\ii\dfrac\Delta{\Omega_R}s_t&-\ii\dfrac{\Omega_1}{\Omega_R}s_t\\[4pt]
     -\ii\dfrac{\Omega_1}{\Omega_R}s_t&c_t+\ii\dfrac\Delta{\Omega_R}s_t
   \end{pmatrix}.}
\end{equation}
作用于初态即取第一列；再乘 $V^\dagger$ 回到实验室系：
\begin{align}
  \ket{\psi_r(t)}&=\begin{pmatrix}
    c_t-\ii(\Delta/\Omega_R)s_t\\-\ii(\Omega_1/\Omega_R)s_t
    \end{pmatrix},\\
  \boxed{\ket{\psi(t)}}&=
   \boxed{\begin{pmatrix}
     \ee^{-\ii\omega t/2}\bigl[c_t-\ii(\Delta/\Omega_R)s_t\bigr]\\[4pt]
     -\ii\ee^{\ii\omega t/2}(\Omega_1/\Omega_R)s_t
   \end{pmatrix}.}
\end{align}
$t=0$ 恢复初态。两个分量的模平方之和为
$c_t^2+(\Delta^2+\Omega_1^2)s_t^2/\Omega_R^2=1$。若同时令 $\Delta=\Omega_1=0$，应直接用 $H_r=0$、$U_r=I$；实验室态为 $\ee^{-\ii\omega t/2}\ket{z,+}$，而非使用带 $1/\Omega_R$ 的表达式。

\textbf{3. 翻转概率与磁矩。}记实验室系分量为 $a_t,b_t$。相位因子不改变模平方，故
\begin{equation}
  \boxed{P_{+\to-}=|b_t|^2
    =\frac{\Omega_1^2}{\Omega_R^2}\sin^2\frac{\Omega_Rt}{2}.}
\end{equation}
利用 $|a_t|^2+|b_t|^2=1$ 和 $2\sin^2(u/2)=1-\cos u$，
\begin{align}
  \expect{S_z}_t
    &=\frac\hbar2(1-2|b_t|^2)
      =\frac\hbar2\frac{\Delta^2+\Omega_1^2\cos\Omega_Rt}{\Omega_R^2},\\
  \boxed{\expect{\mu_z}_t}
    &=\boxed{\frac{\gamma\hbar}{2}
       \frac{\Delta^2+\Omega_1^2\cos\Omega_Rt}{\Delta^2+\Omega_1^2}.}
\end{align}
有效轴 $\hat{\mathbf n}=(\Omega_1,0,\Delta)/\Omega_R$ 满足 $\cos\alpha=\Delta/\Omega_R$，因此该式正是
\begin{equation}
  \expect{\mu_z}_t=\expect{\mu_z}_0
    (\cos^2\alpha+\sin^2\alpha\cos\Omega_Rt),\qquad
  \expect{\mu_z}_0=\frac{\gamma\hbar}{2}<0.
\end{equation}
这与教材的经典旋转几何结构一致；负的初始磁矩分量不能替换成正的磁矩大小。

\textbf{4. 共振态与脉冲。}共振时 $\Delta=0$，旋转系态为
$\bigl(\cos(\Omega_1t/2),-\ii\sin(\Omega_1t/2)\bigr)^{\mathsf T}$，即绕 $+x$ 旋转。其平均方向为 $(0,-\sin\Omega_1t,\cos\Omega_1t)$；回到实验室系还要绕 $+z$ 旋转 $\omega t$。例如在 $0\le\Omega_1t\le\pi$ 内，实验室态与 \S 14.3 的方向旋量比较，极角为 $\Omega_1t$、方位角为 $\omega t-\pi/2$，只差整体相位。

共振翻转概率为 $\sin^2(\Omega_1t/2)$，第一次达到 $1$ 的时刻为
\begin{equation}
  \boxed{t_{\rm flip}=\frac\pi{\Omega_1}.}
\end{equation}
这就是 $\pi$ 脉冲的时长；到达该时刻后关闭横向场，留下的 $B_0$ 只使 $\ket{z,-}$ 获得整体相位。若一直驱动，仍会继续振荡。

固定 $\Delta\ne0$ 时，
\begin{equation}
  \boxed{P_{\max}=\frac{\Omega_1^2}{\Delta^2+\Omega_1^2}<1.}
\end{equation}
所以仅延长恒定幅度、恒定失谐的驱动时间不能完全翻转。上述结论来自给定理想磁场下的精确量子演化，不包含环境弛豫。
\end{checkedsolution}

\exercisequestion{QM-C14-S05-E01}{教材正文示范：自旋加倍与氢原子的磁场分裂}
\textbf{来源：}印刷 pp.~397--398，式 (14.5.2)--(14.5.7)，无独立题号。对应\relatedtopic{QM-C14-S05-T01}{联合态空间}与\relatedtopic{QM-C14-S05-T02}{磁场分裂}。

\textbf{问题。}先仅考虑氢原子库仑 Hamiltonian，再加入沿 $+z$ 的均匀磁场。取电子 $g=2$，忽略质子磁矩、精细与超精细结构及 $B^2$ 项。说明加入自旋后的能量与简并度，并求 $n=1,2$ 在该线性磁场模型中的分裂与剩余简并。

\begin{checkedsolution}
\textbf{1. 无场。}若 $H_C\psi_{nlm_l}=-(\mathrm{Ry}/n^2)\psi_{nlm_l}$，两种正交旋量为
\begin{equation}
  \Psi_{nlm_l,+}=\begin{pmatrix}\psi_{nlm_l}\\0\end{pmatrix},\qquad
  \Psi_{nlm_l,-}=\begin{pmatrix}0\\\psi_{nlm_l}\end{pmatrix}.
\end{equation}
$H_C$ 对两个分量作用相同，故能量均为 $-\mathrm{Ry}/n^2$。固定 $n$ 时，空间态数为 $\sum_{l=0}^{n-1}(2l+1)=n^2$，所以
\begin{equation}
  \boxed{g_n=2n^2.}
\end{equation}
两个态虽有相同空间密度，但自旋正交；任意自旋叠加仍可有相同的确定能量，却未必有确定 $S_z$。

\textbf{2. 加磁场。}由电子轨道及自旋磁矩，
\begin{equation}
  H=H_C+\frac{\mu_BB}{\hbar}(L_z+2S_z),\qquad
  E=-\frac{\mathrm{Ry}}{n^2}+\mu_BB(m_l+2m_s).
\end{equation}
对 $n=1$，只能 $l=m_l=0$，而 $m_s=\pm1/2$，故
\begin{equation}
  \boxed{E_{1,\pm}=-\mathrm{Ry}\pm\mu_BB,\qquad
    E_{1,+}-E_{1,-}=2\mu_BB.}
\end{equation}
电子自旋沿场时磁矩反向，对应较高能量。

对 $n=2$，$l=0$ 提供两个态，$l=1$ 提供六个态。记 $q=m_l+2m_s$：
\begin{center}
\small
\begin{tabularx}{\textwidth}{@{}cXcc@{}}
  \toprule
  $q$ & $(l,m_l,m_s)$ & 能量 & 简并度 \\
  \midrule
  $2$ & $(1,1,1/2)$ & $-\mathrm{Ry}/4+2\mu_BB$ & $1$ \\
  $1$ & $(0,0,1/2)$、$(1,0,1/2)$ & $-\mathrm{Ry}/4+\mu_BB$ & $2$ \\
  $0$ & $(1,1,-1/2)$、$(1,-1,1/2)$ & $-\mathrm{Ry}/4$ & $2$ \\
  $-1$ & $(0,0,-1/2)$、$(1,0,-1/2)$ & $-\mathrm{Ry}/4-\mu_BB$ & $2$ \\
  $-2$ & $(1,-1,-1/2)$ & $-\mathrm{Ry}/4-2\mu_BB$ & $1$ \\
  \bottomrule
\end{tabularx}
\end{center}
因此八个态分到五档能量，态数仍为 $1+2+2+2+1=8$。$q$ 相同只表示磁场位移相同；还须比较 $n$ 才能判断总能量是否相同。
\end{checkedsolution}

\exercisequestion{QM-C14-S05-E02}{教材习题 14.5.2：磁场修正的数量级}
\textbf{对应知识点：}\relatedtopic{QM-C14-S05-T02}{线性磁场近似}。

\textbf{题目。}对氢原子 $n=1$ 态施加 $B=1000\,\mathrm{kG}$：
\begin{enumerate}
  \item 估计两个自旋能级的间距相对于未扰动基态能量绝对值的大小。
  \item 用半径为 $a_0$ 的经典轨道作尺度估计，比较被忽略的 $B^2$ 项与保留的线性磁耦合项，并估计忽略二次项何时不再合理。
\end{enumerate}
计算统一用 SI 制：$1000\,\mathrm{kG}=100\,\mathrm T$，$\mathrm{Ry}=13.6\,\mathrm{eV}$，$\mu_B=5.79\times10^{-5}\,\mathrm{eV/T}$，$a_0=5.29\times10^{-11}\,\mathrm m$，$e=1.60\times10^{-19}\,\mathrm C$，$\hbar=1.055\times10^{-34}\,\mathrm{J\,s}$。

\begin{checkedsolution}
\textbf{1. 能级间距。}由 $E_{1,\pm}=-\mathrm{Ry}\pm\mu_BB$，
\begin{align}
  \Delta E&=2\mu_BB=1.158\times10^{-2}\,\mathrm{eV},\\
  \boxed{\frac{\Delta E}{|E_{C,1}|}}
    &=\boxed{\frac{2\mu_BB}{\mathrm{Ry}}
       \simeq8.5\times10^{-4}=0.085\%.}
\end{align}
这里比较两个能级的间距，不是某一支的位移 $\mu_BB$。

\textbf{2. 二次项。}在对称规范 $\mathbf A=\tfrac B2(-Y,X,0)$ 下，SI 制最小耦合给出
\begin{equation}
  H_{B^2}=\frac{e^2\mathbf A^2}{2m_e}
    =\frac{e^2B^2}{8m_e}(X^2+Y^2).
\end{equation}
取垂直磁场的轨道尺度为 $a_0$，即 $\expect{X^2+Y^2}\sim a_0^2$。这是题目指定的粗估，不把 $1s$ 态理解为确定的经典圆轨道。与单个态的线性修正尺度比较：
\begin{equation}
  E_{B^2}\sim\frac{e^2B^2a_0^2}{8m_e},\qquad
  E_B\sim\mu_BB=\frac{e\hbar B}{2m_e},\qquad
  \boxed{\frac{E_{B^2}}{E_B}\sim\frac{eBa_0^2}{4\hbar}.}
\end{equation}
在 $B=100\,\mathrm T$ 时，
\begin{equation}
  \boxed{\frac{E_{B^2}}{E_B}\sim1.1\times10^{-4}.}
\end{equation}
二次项比线性项小约四个数量级。令两项达到同一数量级，得到
\begin{equation}
  \boxed{B_c\sim\frac{4\hbar}{ea_0^2}
    \simeq9.4\times10^5\,\mathrm T\sim10^6\,\mathrm T.}
\end{equation}
真正的忽略条件是 $eBa_0^2/(4\hbar)\ll1$，不是 $B<B_c$ 就一概足够。$B_c$ 是固定原子尺度下的估计，并非突然失效的精确临界点；强场还会改变波函数尺度。采用不同轨道几何或精确的 $1s$ 二阶矩会改变数值因子，不改变这里的数量级判断。
\end{checkedsolution}

\exercisequestion{QM-C14-S05-E03}{教材习题 14.5.3：串联 SG 筛选器}
\textbf{对应知识点：}\relatedtopic{QM-C14-S05-T03}{分束、条件筛选与相干性}。

\textbf{题目。}自旋 $1/2$ 粒子束沿 $y$ 方向通过串联 SG 装置。第一个只通过 $z+$，第二个的筛选轴转到 $x$ 方向，只通过 $x+$。所有通过率均以离开第一个筛选器的粒子数为分母。
\begin{enumerate}
  \item 求通过第二个筛选器的比例。
  \item 再加一个只通过 $z+$ 的筛选器，求最后的总通过率。
  \item 改为中间不阻挡任何一束，即保留 $x+$、$x-$，最后只通过 $z-$，求总通过率。
\end{enumerate}
\textbf{第三问的条件补充。}教材只写中间不阻挡，未明确两条空间路径的相干重合方式。本次先按两束仍可区分、都进入最后装置且计数相加求解，再对照相干逆转分束、恢复同一空间模式与原相对相位的情况。

\begin{checkedsolution}
采用相位约定
\begin{equation}
  \ket{x,+}=\frac{\ket{z,+}+\ket{z,-}}{\sqrt2},\qquad
  \ket{x,-}=\frac{\ket{z,+}-\ket{z,-}}{\sqrt2}.
\end{equation}
因此各个 $x$ 本征态在 $z$ 方向的两种测量概率均为 $1/2$。

\textbf{1. 第二个筛选器。}输入为 $\ket{z,+}$，
\begin{equation}
  \boxed{P_2=|\braket{x,+}{z,+}|^2=\frac12.}
\end{equation}
以通过为条件，输出自旋态为 $\ket{x,+}$。

\textbf{2. 最后只通过 $z+$。}最后一级的条件通过率为 $1/2$，但中间已丢弃一半粒子，所以
\begin{equation}
  \boxed{P_3=|\braket{x,+}{z,+}|^2
    |\braket{z,+}{x,+}|^2=\frac14.}
\end{equation}
不能把“相对进入最后一级的粒子”与“相对离开第一级的粒子”混为一谈。

\textbf{3. 保留两条可区分路径。}经过 $x+$ 与 $x-$ 两支而最后通过 $z-$ 的贡献各为 $1/4$，因此
\begin{equation}
  \boxed{P_{z-}=\frac12\cdot\frac12+\frac12\cdot\frac12=\frac12.}
\end{equation}
这里并不要求中间已经发生投影测量。用归一化空间模式 $\ket{f_+},\ket{f_-}$ 保留完整态：
\begin{equation}
  \ket\Psi=\frac1{\sqrt2}
    \left(\ket{f_+}\ket{x,+}+\ket{f_-}\ket{x,-}\right).
\end{equation}
投到 $z-$ 后的未归一化空间振幅为
\begin{equation}
  (I_o\otimes\bra{z,-})\ket\Psi
    =\frac12(\ket{f_+}-\ket{f_-}).
\end{equation}
所以，若最后一级对两支作相同的理想自旋筛选并统计全部空间输出，
\begin{equation}
  \boxed{P_{z-}=\frac14\|\ket{f_+}-\ket{f_-}\|^2
    =\frac12\left(1-\operatorname{Re}\braket{f_+}{f_-}\right).}
\end{equation}
保持分离时，$\braket{f_+}{f_-}\simeq0$，即得 $1/2$；正交的路径记录足以消除这一统计中的交叉项，整体态仍可保持纯态。

\textbf{相干合束对照。}若通过相干操作把两支恢复为相同模式且相对相位恢复，即 $\ket{f_+}=\ket{f_-}=\ket f$，则
\begin{equation}
  \ket\Psi=\ket f\,\frac{\ket{x,+}+\ket{x,-}}{\sqrt2}
    =\ket f\ket{z,+},\qquad \boxed{P_{z-}=0.}
\end{equation}
仅让两束方向平行或都不被阻挡，不足以得到这个结论；相同空间模式还须有正确的相对相位。这个对照补清原题的装置条件，不把“不阻挡”直接等同于恢复初态。
\end{checkedsolution}
